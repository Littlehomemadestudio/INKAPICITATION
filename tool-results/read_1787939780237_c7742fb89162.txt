    29→  initFonts() {

    33→      const sans = style.getPropertyValue('--font-geist-sans').trim();

    41→  draw(ctx: CanvasRenderingContext2D, game: Game, dpr: number) {
    42→    const cam = game.camera;
    43→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    44→    // dark table around the sheet
    45→    ctx.fillStyle = '#161513';
    46→    ctx.fillRect(0, 0, cam.viewW, cam.viewH);
    47→
    48→    cam.applyTransform(ctx, dpr);
    49→
    50→    game.terrainRenderer.drawBase(ctx, cam, game.effects.scars);
    51→    game.terrainRenderer.drawFeatures(ctx, cam);
    52→    game.effects.drawCraters(ctx, cam);
    53→    game.effects.drawRubble(ctx, cam);
    54→
    55→    const hovers: HoverLabel[] = [];
    56→    this.drawSectors(ctx, game, cam);
    57→    this.drawAssembly(ctx, game, cam);
    58→    this.drawObjectives(ctx, game, cam);
    59→    this.drawFactories(ctx, game, cam, hovers);
    60→    this.drawWrecks(ctx, game, cam);
    61→    this.drawUnits(ctx, game, cam, hovers);
    62→    game.projectiles.draw(ctx);
    63→    game.effects.drawCore(ctx);
    64→    game.effects.drawOrderMarkers(ctx);
    65→    game.effects.drawSmoke(ctx);
    66→    game.terrainRenderer.drawFurniture(ctx);
    67→
    68→    // screen space
    69→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    70→    this.drawPaperGrain(ctx, game, cam);
    71→    this.drawSelectionBox(ctx, game);
    72→    this.drawHoverLabels(ctx, hovers);
    73→    this.drawCursorMode(ctx, game);
    74→  }

    76→  /** constant screen-size paper grain, anchored to world position */
    77→  private drawPaperGrain(ctx: CanvasRenderingContext2D, game: Game, cam: Camera) {
    78→    const strength = clamp((cam.zoom - 0.35) * 1.1, 0, 0.55);
    79→    if (strength <= 0.02) return;
    80→    if (!this.grainPattern) {
    81→      this.grainPattern = ctx.createPattern(game.terrainRenderer.grain, 'repeat');
    82→      if (!this.grainPattern) return;
    83→    }
    84→    const ox = ((-cam.x * cam.zoom) % 256 + 256) % 256;
    85→    const oy = ((-cam.y * cam.zoom) % 256 + 256) % 256;
    86→    ctx.save();
    87→    ctx.translate(ox - 256, oy - 256);
    88→    ctx.globalAlpha = strength;
    89→    ctx.fillStyle = this.grainPattern;
    90→    ctx.fillRect(0, 0, cam.viewW + 512, cam.viewH + 512);
    91→    ctx.restore();
    92→  }

    96→  private drawSectors(ctx: CanvasRenderingContext2D, game: Game, cam: Camera) {
    97→    if (cam.zoom > 1.6) return; // at close range the ground speaks for itself
    98→    for (const s of game.economy.sectors) {
    99→      const inView =
   100→        s.pos.x + s.radius > cam.viewX &&
   101→        s.pos.x - s.radius < cam.viewX + cam.viewW &&
   102→        s.pos.y + s.radius > cam.viewY &&
   103→        s.pos.y - s.radius < cam.viewY + cam.viewH;
   104→      if (!inView) continue;
   105→      this.drawSector(ctx, s, cam);
   106→    }
   107→  }

   109→  private drawSector(ctx: CanvasRenderingContext2D, s: Sector, cam: Camera) {
   110→    const strong = cam.zoom < 0.7;
   111→    ctx.save();
   112→    // boundary — a surveyed area, not a video game circle
   113→    ctx.strokeStyle =
   114→      s.control === 'FRIEND' ? 'rgba(20,17,12,0.55)' : s.control === 'ENEMY' ? 'rgba(80,74,62,0.5)' : 'rgba(110,104,90,0.38)';
   115→    ctx.lineWidth = Math.max(1, 1 / cam.zoom);
   116→    ctx.setLineDash([16, 12]);
   117→    ctx.beginPath();
   118→    ctx.arc(s.pos.x, s.pos.y, s.radius, 0, Math.PI * 2);
   119→    ctx.stroke();
   120→    ctx.setLineDash([]);
   121→    // inner tick ring at quarter radius
   122→    if (strong) {
   123→      ctx.globalAlpha = 0.5;
   124→      ctx.setLineDash([3, 14]);
   125→      ctx.beginPath();
   126→      ctx.arc(s.pos.x, s.pos.y, s.radius * 0.55, 0, Math.PI * 2);
   127→      ctx.stroke();
   128→      ctx.setLineDash([]);
   129→      ctx.globalAlpha = 1;
   130→    }
   131→
   132→    // capture progress — an arc filling as the ground changes hands
   133→    if (s.capturing && s.captureT > 0.2) {
   134→      const frac = clamp(s.captureT / s.captureTime, 0, 1);
   135→      ctx.strokeStyle = s.capturing === 'FRIEND' ? '#141210' : '#5a544a';
   136→      ctx.lineWidth = Math.max(2.2, 2.6 / cam.zoom);
   137→      ctx.beginPath();
   138→      ctx.arc(s.pos.x, s.pos.y, s.radius * 0.72, -Math.PI / 2, -Math.PI / 2 + frac * Math.PI * 2);
   139→      ctx.stroke();
   140→    }
   141→
   142→    // ownership mark — square for friendly, hollow for enemy, dot for neutral
   143→    const ms = Math.max(9, 12 / cam.zoom);
   144→    if (s.control === 'FRIEND') {
   145→      ctx.fillStyle = 'rgba(20,17,12,0.92)';
   146→      ctx.fillRect(s.pos.x - ms / 2, s.pos.y - ms / 2, ms, ms);
   147→      ctx.fillStyle = '#f3f1ea';
   148→      ctx.fillRect(s.pos.x - ms * 0.14, s.pos.y - ms * 0.14, ms * 0.28, ms * 0.28);
   149→    } else if (s.control === 'ENEMY') {
   150→      ctx.strokeStyle = 'rgba(70,64,52,0.9)';
   151→      ctx.lineWidth = Math.max(1.4, 1.6 / cam.zoom);
   152→      ctx.strokeRect(s.pos.x - ms / 2, s.pos.y - ms / 2, ms, ms);
   153→      ctx.fillStyle = 'rgba(70,64,52,0.75)';
   154→      ctx.fillRect(s.pos.x - ms * 0.14, s.pos.y - ms * 0.14, ms * 0.28, ms * 0.28);
   155→    } else {
   156→      ctx.strokeStyle = 'rgba(96,90,76,0.8)';
   157→      ctx.lineWidth = Math.max(1.1, 1.2 / cam.zoom);
   158→      ctx.beginPath();
   159→      ctx.arc(s.pos.x, s.pos.y, ms * 0.42, 0, Math.PI * 2);
   160→      ctx.stroke();
   161→    }
   162→
   163→    // designation
   164→    if (cam.zoom > 0.2) {
   165→      const size = clamp(12 / cam.zoom + 5, 11, 26);
   166→      ctx.font = `600 ${size}px ${this.sansFont}`;
   167→      ctx.textAlign = 'center';
   168→      ctx.textBaseline = 'middle';
   169→      const label = `${s.name} · +${s.income.toFixed(1)}`;
   170→      ctx.strokeStyle = 'rgba(243,241,234,0.85)';
   171→      ctx.lineWidth = size * 0.22;
   172→      ctx.strokeText(label, s.pos.x, s.pos.y - ms * 1.4);
   173→      ctx.fillStyle = 'rgba(28,25,19,0.85)';
   174→      ctx.fillText(label, s.pos.x, s.pos.y - ms * 1.4);
   175→    }
   176→    ctx.restore();
   177→  }

   205→  private drawObjectives(ctx: CanvasRenderingContext2D, game: Game, cam: Camera) {
   206→    for (const obj of game.objectives as ObjectiveState[]) {
   207→      const size = 95;
   208→      ctx.save();
   209→      ctx.strokeStyle = '#141210';
   210→      ctx.lineWidth = Math.max(1.1, 1.1 / cam.zoom);
   211→      if (!obj.secured) ctx.setLineDash([10, 7]);
   212→      // corner brackets
   213→      const draw = (s: number, gap: number) => {
   214→        const l = s * 0.42;
   215→        for (const [sx, sy] of [[1, 1], [1, -1], [-1, 1], [-1, -1]] as const) {
   216→          ctx.beginPath();
   217→          ctx.moveTo(obj.pos.x + sx * s - sx * l, obj.pos.y + sy * (s + gap));
   218→          ctx.lineTo(obj.pos.x + sx * s, obj.pos.y + sy * (s + gap));
   219→          ctx.lineTo(obj.pos.x + sx * s, obj.pos.y + sy * (s + gap) - sy * l);
   220→          ctx.stroke();
   221→        }
   222→      };
   223→      draw(size, 26);
   224→      if (obj.primary) draw(size + 16, 26);
   225→      ctx.setLineDash([]);
   226→      // leader line from bracket to label
   227→      ctx.strokeStyle = 'rgba(24,21,15,0.45)';
   228→      ctx.lineWidth = Math.max(0.8, 0.8 / cam.zoom);
   229→      ctx.beginPath();
   230→      ctx.moveTo(obj.pos.x, obj.pos.y - size - 4);
   231→      ctx.lineTo(obj.pos.x, obj.pos.y - size - 16);
   232→      ctx.stroke();
   233→      // label
   234→      const labelSize = clamp(15 / cam.zoom + 8, 12, 30);
   235→      ctx.font = `600 ${labelSize}px ${this.sansFont}`;
   236→      ctx.textAlign = 'center';
   237→      ctx.fillStyle = obj.secured ? 'rgba(24,21,15,0.95)' : 'rgba(24,21,15,0.8)';
   238→      ctx.strokeStyle = 'rgba(243,241,234,0.9)';
   239→      ctx.lineWidth = labelSize * 0.22;
   240→      const status = obj.secured ? 'SECURED' : 'HOSTILE';
   241→      const text = `${obj.name} — ${status}`;
   242→      ctx.strokeText(text, obj.pos.x, obj.pos.y - size - 22);
   243→      ctx.fillText(text, obj.pos.x, obj.pos.y - size - 22);
   244→      ctx.restore();
   245→    }

   250→  private drawFactories(ctx: CanvasRenderingContext2D, game: Game, cam: Camera, hovers: HoverLabel[]) {
   251→    for (const u of game.units) {
   252→      if (u.def.kind !== 'FACTORY') continue;
   253→      if (u.dead) continue;
   254→      const zx = u.x;
   255→      const zy = u.y;
   256→      const halfW = u.def.length * 0.62;
   257→      const halfH = u.def.width * 0.85;
   258→
   259→      ctx.save();
   260→      ctx.translate(zx, zy);
   261→      ctx.rotate(0.08);
   262→
   263→      // perimeter — surveyed boundary of the installation
   264→      const friendly = u.factoryCtl === 'FRIEND';
   265→      const neutral = u.factoryCtl === 'NEUTRAL';
   266→      ctx.strokeStyle = friendly
   267→        ? 'rgba(20,17,12,0.8)'
   268→        : neutral
   269→          ? 'rgba(104,98,84,0.65)'
   270→          : 'rgba(76,70,58,0.8)';
   271→      ctx.lineWidth = Math.max(1.3, 1.5 / cam.zoom);
   272→      ctx.setLineDash([14, 9]);
   273→      ctx.strokeRect(-halfW, -halfH, halfW * 2, halfH * 2);
   274→      ctx.setLineDash([]);
   275→
   276→      // capture progress arc across the works
   277→      if (u.capturing && u.captureT > 0.2) {
   278→        const frac = clamp(u.captureT / 7, 0, 1);
   279→        ctx.strokeStyle = u.capturing === 'FRIEND' ? '#141210' : '#5a544a';
   280→        ctx.lineWidth = Math.max(2.6, 3 / cam.zoom);
   281→        ctx.beginPath();
   282→        ctx.arc(0, 0, Math.max(halfW, halfH) * 1.12, -Math.PI / 2, -Math.PI / 2 + frac * Math.PI * 2);
   283→        ctx.stroke();
   284→      }
   285→
   286→      // control pennant — pole and flag at the works corner
   287→      const px = halfW * 0.8;
   288→      const py = -halfH * 0.8;
   289→      ctx.strokeStyle = '#26221b';
   290→      ctx.lineWidth = Math.max(1, 1.1 / cam.zoom);
   291→      ctx.beginPath();
   292→      ctx.moveTo(px, py);
   293→      ctx.lineTo(px, py - 16);
   294→      ctx.stroke();
   295→      ctx.beginPath();
   296→      if (friendly) {
   297→        ctx.fillStyle = '#141210';
   298→        ctx.moveTo(px, py - 16);
   299→        ctx.lineTo(px + 11, py - 12.5);
   300→        ctx.lineTo(px, py - 9);
   301→        ctx.closePath();
   302→        ctx.fill();
   303→      } else if (neutral) {
   304→        ctx.strokeStyle = 'rgba(96,90,76,0.9)';
   305→        ctx.lineWidth = Math.max(1, 1.1 / cam.zoom);
   306→        ctx.moveTo(px, py - 16);
   307→        ctx.lineTo(px + 11, py - 12.5);
   308→        ctx.lineTo(px, py - 9);
   309→        ctx.closePath();
   310→        ctx.stroke();
   311→      } else {
   312→        ctx.fillStyle = 'rgba(82,76,64,0.9)';
   313→        ctx.moveTo(px, py - 16);
   314→        ctx.lineTo(px + 11, py - 12.5);
   315→        ctx.lineTo(px, py - 9);
   316→        ctx.closePath();
   317→        ctx.fill();
   318→      }
   319→
   320→      // designation plate
   321→      if (cam.zoom > 0.24) {
   322→        const size = clamp(11 / cam.zoom + 5, 10, 24);
   323→        ctx.font = `600 ${size}px ${this.sansFont}`;
   324→        ctx.textAlign = 'center';
   325→        ctx.textBaseline = 'middle';
   326→        const status = friendly ? 'HOLDING · +5 INK/S' : neutral ? 'UNCLAIMED · +5 INK/S' : 'ENEMY HELD · +5 INK/S';
   327→        const label = `${u.callsign} — ${status}`;
   328→        ctx.strokeStyle = 'rgba(243,241,234,0.88)';
   329→        ctx.lineWidth = size * 0.22;
   330→        ctx.strokeText(label, 0, -halfH - size * 0.9);
   331→        ctx.fillStyle = friendly ? 'rgba(24,21,15,0.92)' : 'rgba(46,42,34,0.85)';
   332→        ctx.fillText(label, 0, -halfH - size * 0.9);
   333→      }
   334→
   335→      // damage: soot creeping across the works
   336→      if (u.hp < u.def.hp * 0.7) {
   337→        const dmg = 1 - u.hp / u.def.hp;
   338→        ctx.fillStyle = `rgba(22,18,12,${0.18 * dmg})`;
   339→        ctx.fillRect(-halfW, -halfH, halfW * 2, halfH * 2);
   340→      }
   341→      ctx.restore();
   342→
   343→      if (game.input.hoverUnit === u) {
   344→        const s = cam.worldToScreen(zx, zy);
   345→        hovers.push({
   346→          sx: s.x,
   347→          sy: s.y,
   348→          hostile: !friendly,
   349→          lines: [
   350→            `${u.callsign} · INK WORKS`,
   351→            friendly ? 'FRIENDLY — PRODUCING' : neutral ? 'UNCLAIMED — CAPTURE TO DRAW INK' : 'ENEMY — CAPTURE OR DESTROY',
   352→            `INTEGRITY ${Math.ceil((u.hp / u.def.hp) * 100)}%`,
   353→          ],
   354→        });
   355→      }
   356→    }
   357→  }

   361→  private drawWrecks(ctx: CanvasRenderingContext2D, game: Game, cam: Camera) {
   362→    const zoom = cam.zoom;
   363→    const detail = zoom < 0.28 ? 0 : zoom < 0.85 ? 1 : 2;
   364→    for (const w of game.effects.wrecks) {
   365→      // strategic ink blot — wrecks must register from the overview
   366→      if (zoom < 0.5) {
   367→        const r = Math.max(9, 13 / zoom);
   368→        const g = ctx.createRadialGradient(w.x, w.y, 0, w.x, w.y, r);
   369→        g.addColorStop(0, 'rgba(20,17,11,0.5)');
   370→        g.addColorStop(1, 'rgba(20,17,11,0)');
   371→        ctx.fillStyle = g;
   372→        ctx.beginPath();
   373→        ctx.arc(w.x, w.y, r, 0, Math.PI * 2);
   374→        ctx.fill();
   375→      }
   376→      const age = game.time - w.born;
   377→      const alpha = clamp(1 - age / 900, 0.62, 1);
   378→      ctx.save();
   379→      ctx.globalAlpha = alpha;
   380→      ctx.translate(w.x, w.y);
   381→      ctx.rotate(w.angle);
   382→      drawVehicle(ctx, {
   383→        type: w.type as never,
   384→        style: WRECK_STYLE,
   385→        detail,
   386→        turretAngle: 0,
   387→        radarAngle: 0,
   388→        wreck: true,
   389→        noTurret: true,
   390→      });
   391→      ctx.restore();
   392→      // tossed turret
   393→      if (w.turretToss) {
   394→        ctx.save();
   395→        ctx.globalAlpha = alpha;
   396→        ctx.translate(w.turretToss.x, w.turretToss.y);
   397→        ctx.rotate(w.turretToss.angle);
   398→        ctx.fillStyle = '#232019';
   399→        ctx.beginPath();
   400→        ctx.ellipse(0, 0, 3.4, 3.0, 0, 0, Math.PI * 2);
   401→        ctx.fill();
   402→        ctx.fillRect(2.6, -0.35, 4.4, 0.7);
   403→        if (detail >= 1) {
   404→          ctx.fillStyle = '#2e2a22';
   405→          ctx.beginPath();
   406→          ctx.arc(-0.6, 0.7, 0.8, 0, Math.PI * 2);
   407→          ctx.fill();
   408→        }
   409→        ctx.restore();
   410→      }
   411→    }
   412→  }

   416→  /** cover state of a friendly unit vs the nearest known threat */
   417→  private coverOf(u: Unit, game: Game, x: number, y: number): { value: number; type: string; threatX: number; threatY: number } | null {
   418→    // the shooter is the threat that matters; otherwise nearest contact
   419→    let tx = 0;
   420→    let ty = 0;
   421→    let td = Infinity;
   422→    if (u.lastAttacker && !u.lastAttacker.dead && game.time - u.lastAttackedT < 20) {
   423→      tx = u.lastAttacker.x;
   424→      ty = u.lastAttacker.y;
   425→      td = 0;
   426→    } else {
   427→      for (const e of game.units) {
   428→        if (e.dead || e.faction !== 'ENEMY' || e.isAir) continue;
   429→        if (e.intel === 'HIDDEN' && e.def.kind !== 'FACTORY') continue;
   430→        const d = Math.hypot(e.x - x, e.y - y);
   431→        if (d < td && d < 1500) {
   432→          td = d;
   433→          tx = e.x;
   434→          ty = e.y;
   435→        }
   436→      }
   437→    }
   438→    if (td === Infinity) return null;
   439→    const r = coverFrom(game.simCtx(), x, y, tx, ty);
   440→    return { value: r.value, type: r.type, threatX: tx, threatY: ty };
   441→  }

   443→  private drawUnits(ctx: CanvasRenderingContext2D, game: Game, cam: Camera, hovers: HoverLabel[]) {
   444→    const zoom = cam.zoom;
   445→    const detail = zoom < 0.28 ? 0 : zoom < 0.85 ? 1 : 2;
   446→    // strategic readability: units never shrink below a legible stamp
   447→    const minPx = 15;
   448→    const scaleFor = (u: Unit) => Math.min(4.2, Math.max(1, minPx / (zoom * u.def.length)));
   449→
   450→    // ground units first, aircraft on top
   451→    const ground: Unit[] = [];
   452→    const air: Unit[] = [];
   453→    for (const u of game.units) {
   454→      if (u.dead || u.def.kind === 'FACTORY') continue;
   455→      if (u.isAir) {
   456→        if (u.airState === 'STANDBY' || u.airState === 'REARM') continue;
   457→        air.push(u);
   458→      } else {
   459→        if (u.faction === 'ENEMY' && u.intel === 'HIDDEN') continue;
   460→        ground.push(u);
   461→      }
   462→    }
   463→
   464→    for (const u of ground) {
   465→      const selected = game.input.selection.includes(u);
   466→      const ghost = u.faction === 'ENEMY' && u.intel === 'GHOST';
   467→      const gx = ghost ? u.knownX : u.x;
   468→      const gy = ghost ? u.knownY : u.y;
   469→      const es = scaleFor(u);
   470→
   471→      ctx.save();
   472→      ctx.translate(gx, gy);
   473→
   474→      if (ghost) {
   475→        ctx.globalAlpha = 0.34;
   476→        ctx.strokeStyle = '#565046';
   477→        ctx.lineWidth = 1.2;
   478→        ctx.setLineDash([4, 4]);
   479→        ctx.beginPath();
   480→        ctx.arc(0, 0, 13 * es, 0, Math.PI * 2);
   481→        ctx.stroke();
   482→        ctx.setLineDash([]);
   483→        ctx.globalAlpha = 1;
   484→      }
   485→
   486→      const style = u.faction === 'FRIEND' ? FRIEND_STYLE : ENEMY_STYLE;
   487→      // paper halo — separates the stamp from terrain shading
   488→      if (es > 1.15 && !ghost) {
   489→        ctx.fillStyle = 'rgba(243,241,234,0.5)';
   490→        ctx.beginPath();
   491→        ctx.ellipse(0, 0, u.def.length * 0.62 * es, u.def.width * 0.85 * es, u.angle, 0, Math.PI * 2);
   492→        ctx.fill();
   493→      }
   494→      // shadow
   495→      if (!ghost && es <= 1.15) {
   496→        ctx.save();
   497→        ctx.translate(2.6, 3.2);
   498→        ctx.rotate(u.angle);
   499→        ctx.fillStyle = 'rgba(25,22,16,0.15)';
   500→        ctx.beginPath();
   501→        ctx.ellipse(0, 0, u.def.length * 0.52, u.def.width * 0.58, 0, 0, Math.PI * 2);
   502→        ctx.fill();
   503→        ctx.restore();
   504→      }
   505→
   506→      ctx.save();
   507→      ctx.scale(es, es);
   508→      ctx.rotate(u.angle);
   509→      drawVehicle(ctx, {
   510→        type: u.def.type,
   511→        style,
   512→        detail,
   513→        turretAngle: u.turretAngle - u.angle,
   514→        radarAngle: u.radarAngle,
   515→        recoil: u.recoil * (u.def.projectile === 'SHELL' || u.def.projectile === 'ARTY' ? 0.8 : 0.3),
   516→        missiles: u.isAir ? u.ammo : undefined,
   517→      });
   518→      ctx.restore();
   519→
   520→      // damage flash
   521→      if (u.damageFlash > 0) {
   522→        ctx.save();
   523→        ctx.rotate(u.angle);
   524→        ctx.globalAlpha = u.damageFlash * 0.5;
   525→        ctx.fillStyle = '#f3f1ea';
   526→        ctx.beginPath();
   527→        ctx.ellipse(0, 0, u.def.length * 0.42 * es, u.def.width * 0.42 * es, 0, 0, Math.PI * 2);
   528→        ctx.fill();
   529→        ctx.restore();
   530→      }
   531→      ctx.restore();
   532→
   533→      // under fire: impacts kicking up around a suppressed unit —
   534→      // the battlefield tells you who is catching it
   535→      if (!ghost && u.suppression > 0.45 && Math.random() < 0.12) {
   536→        const a = Math.random() * Math.PI * 2;
   537→        const d = u.def.length * (0.5 + Math.random() * 0.6);
   538→        game.effects.spawnDust(
   539→          gx + Math.cos(a) * d,
   540→          gy + Math.sin(a) * d,
   541→          Math.cos(a) * 6,
   542→          Math.sin(a) * 6
   543→        );
   544→      }
   545→
   546→      // selection
   547→      if (selected && u.faction === 'FRIEND') {
   548→        ctx.save();
   549→        ctx.translate(gx, gy);
   550→        drawSelectionBrackets(ctx, u.def.length * 0.62 * Math.max(es, 1), 7 / zoom, Math.max(1.2, 1.3 / zoom), '#141210');
   551→        // the cover mark — a bracketed arc on the sheltered side, so the
   552→        // player can read WHY this vehicle is alive over the open field
   553→        const cov = this.coverOf(u, game, gx, gy);
   554→        if (cov && cov.value >= 0.34) {
   555→          const threatA = Math.atan2(cov.threatY - gy, cov.threatX - gx);
   556→          const ca = threatA + Math.PI; // the protected side
   557→          const rr = u.def.length * 0.62 * Math.max(es, 1) + 8 / zoom;
   558→          const tick = Math.max(4.5, 6.5 / zoom);
   559→          ctx.save();
   560→          ctx.rotate(ca);
   561→          // paper backing so the mark survives craters and smoke
   562→          ctx.fillStyle = 'rgba(243,241,234,0.75)';
   563→          ctx.beginPath();
   564→          ctx.arc(0, 0, rr + tick * 0.9, -Math.PI / 2.4, Math.PI / 2.4);
   565→          ctx.arc(0, 0, rr - tick * 0.6, Math.PI / 2.4, -Math.PI / 2.4, true);
   566→          ctx.closePath();
   567→          ctx.fill();
   568→          // the arc itself
   569→          ctx.strokeStyle = '#141210';
   570→          ctx.lineWidth = Math.max(1.8, 2.4 / zoom);
   571→          ctx.lineCap = 'butt';
   572→          ctx.beginPath();
   573→          ctx.arc(0, 0, rr, -Math.PI / 2.6, Math.PI / 2.6);
   574→          ctx.stroke();
   575→          // three ticks pointing into the cover
   576→          for (let i = -1; i <= 1; i++) {
   577→            const ta = i * 0.38;
   578→            const ax = Math.cos(ta) * rr;
   579→            const ay = Math.sin(ta) * rr;
   580→            const nx = Math.cos(ta);
   581→            const ny = Math.sin(ta);
   582→            ctx.beginPath();
   583→            ctx.moveTo(ax - ny * tick * 0.34, ay + nx * tick * 0.34);
   584→            ctx.lineTo(ax + nx * tick * 0.66, ay + ny * tick * 0.66);
   585→            ctx.lineTo(ax + ny * tick * 0.34, ay - nx * tick * 0.34);
   586→            ctx.stroke();
   587→          }
   588→          ctx.restore();
   589→        }
   590→        // the tactical envelope — what this vehicle can SEE (faint
   591→        // ring) and REACH with its gun (hard ring). The tactics are
   592→        // laid on the terrain itself: stand here and this is your
   593→        // world; move there and the rings tell you what changes.
   594→        if (!u.isAir) {
   595→          const visR = u.def.vision * (1 - 0.3 * u.suppression);
   596→          ctx.strokeStyle = 'rgba(20,18,16,0.16)';
   597→          ctx.setLineDash([12 / zoom, 10 / zoom]);
   598→          ctx.lineWidth = Math.max(1, 1 / zoom);
   599→          ctx.beginPath();
   600→          ctx.arc(gx, gy, visR, 0, Math.PI * 2);
   601→          ctx.stroke();
   602→          if (u.def.range > 40) {
   603→            ctx.strokeStyle = 'rgba(20,18,16,0.38)';
   604→            ctx.setLineDash([16 / zoom, 8 / zoom]);
   605→            ctx.beginPath();
   606→            ctx.arc(gx, gy, u.def.range, 0, Math.PI * 2);
   607→            ctx.stroke();
   608→          }
   609→          ctx.setLineDash([]);
   610→        }
   611→        // command path
   612→        if (u.dest) {
   613→          ctx.strokeStyle = 'rgba(20,17,12,0.5)';
   614→          ctx.lineWidth = Math.max(0.9, 1 / zoom);
   615→          ctx.setLineDash([6, 6]);
   616→          ctx.beginPath();
   617→          ctx.moveTo(gx, gy);
   618→          ctx.lineTo(u.dest.x, u.dest.y);
   619→          ctx.stroke();
   620→          ctx.setLineDash([]);
   621→          ctx.strokeRect(u.dest.x - 3, u.dest.y - 3, 6, 6);
   622→        }
   623→        // hp bar when damaged
   624→        if (u.hp < u.def.hp) {
   625→          const w = Math.max(16, u.def.length * 2.2 * Math.max(es, 1));
   626→          const yy = gy + (u.def.width + 9) * Math.max(es, 1);
   627→          ctx.fillStyle = 'rgba(243,241,234,0.85)';
   628→          ctx.fillRect(gx - w / 2, yy, w, 2.6);
   629→          ctx.fillStyle = '#141210';
   630→          ctx.fillRect(gx - w / 2, yy, (w * u.hp) / u.def.hp, 2.6);
   631→          // suppression — a second, thinner stress bar under the hp
   632→          if (u.suppression > 0.2) {
   633→            ctx.fillStyle = 'rgba(243,241,234,0.7)';
   634→            ctx.fillRect(gx - w / 2, yy + 3.4, w, 1.6);
   635→            ctx.fillStyle = '#5a544a';
   636→            ctx.fillRect(gx - w / 2, yy + 3.4, w * u.suppression, 1.6);
   637→          }
   638→        }
   639→        ctx.restore();
   640→      }
   641→
   642→      // hover label data
   643→      if (game.input.hoverUnit === u) {
   644→        const s = cam.worldToScreen(gx, gy);
   645→        if (u.faction === 'FRIEND') {
   646→          const cov = this.coverOf(u, game, gx, gy);
   647→          const coverLine = !cov
   648→            ? ''
   649→            : cov.value >= 0.65
   650→              ? 'DUG IN'
   651→              : cov.value >= 0.45
   652→                ? 'HARD COVER'
   653→                : cov.value >= 0.34
   654→                  ? 'IN COVER'
   655→                  : cov.type === 'TERRAIN'
   656→                    ? 'DEFILE'
   657→                    : 'EXPOSED';
   658→          hovers.push({
   659→            sx: s.x,
   660→            sy: s.y,
   661→            hostile: false,
   662→            lines: [
   663→              `${u.callsign} · ${u.def.shortName}`,
   664→              u.pinned ? 'PINNED DOWN' : u.suppression > 0.5 ? 'SUPPRESSED' : u.getActivity(),
   665→              coverLine || `GRID ${u.positionGrid()}`,
   666→              ...(coverLine ? [`GRID ${u.positionGrid()}`] : []),
   667→            ],
   668→          });
   669→        } else {
   670→          hovers.push({
   671→            sx: s.x,
   672→            sy: s.y,
   673→            hostile: true,
   674→            lines: [
   675→              `ENEMY · ${u.def.shortName}`,
   676→              ghost ? 'LAST KNOWN POSITION' : u.getActivity(),
   677→              `GRID ${u.positionGrid()}`,
   678→            ],
   679→          });
   680→        }
   681→      }
   682→    }
   683→
   684→    // aircraft
   685→    for (const u of air) {
   686→      const selected = game.input.selection.includes(u);
   687→      const es = scaleFor(u) * 0.7;
   688→      // ground shadow (altitude illusion)
   689→      ctx.save();
   690→      ctx.translate(u.x + 26, u.y + 34);
   691→      ctx.rotate(u.angle);
   692→      ctx.fillStyle = 'rgba(25,22,16,0.10)';
   693→      ctx.beginPath();
   694→      ctx.ellipse(0, 0, u.def.length * 0.5, u.def.width * 0.5, 0, 0, Math.PI * 2);
   695→      ctx.fill();
   696→      ctx.restore();
   697→
   698→      ctx.save();
   699→      ctx.translate(u.x, u.y);
   700→      if (es > 1.1) {
   701→        ctx.fillStyle = 'rgba(243,241,234,0.4)';
   702→        ctx.beginPath();
   703→        ctx.ellipse(0, 0, u.def.length * 0.5 * es, u.def.width * 0.55 * es, u.angle, 0, Math.PI * 2);
   704→        ctx.fill();
   705→      }
   706→      ctx.save();
   707→      ctx.scale(es, es);
   708→      ctx.rotate(u.angle);
   709→      ctx.scale(1, 1 - Math.abs(u.bank) * 0.28);
   710→      drawVehicle(ctx, {
   711→        type: u.def.type,
   712→        style: FRIEND_STYLE,
   713→        detail,
   714→        turretAngle: 0,
   715→        radarAngle: 0,
   716→        missiles: u.ammo,
   717→      });
   718→      ctx.restore();
   719→      ctx.restore();
   720→
   721→      if (selected) {
   722→        ctx.save();
   723→        ctx.translate(u.x, u.y);
   724→        drawSelectionBrackets(ctx, 26, 8 / zoom, Math.max(1.2, 1.3 / zoom), '#141210');
   725→        // patrol orbit
   726→        ctx.strokeStyle = 'rgba(20,17,12,0.3)';
   727→        ctx.lineWidth = Math.max(0.8, 0.9 / zoom);
   728→        ctx.setLineDash([10, 10]);
   729→        ctx.beginPath();
   730→        ctx.arc(u.patrol.x, u.patrol.y, 640, 0, Math.PI * 2);
   731→        ctx.stroke();
   732→        ctx.setLineDash([]);
   733→        ctx.restore();
   734→      }
   735→
   736→      if (game.input.hoverUnit === u) {
   737→        const s = cam.worldToScreen(u.x, u.y);
   738→        hovers.push({
   739→          sx: s.x,
   740→          sy: s.y,
   741→          hostile: false,
   742→          lines: [`${u.callsign} · ${u.def.shortName}`, u.getActivity(), `MSL ${u.ammo}/${u.def.ammo}`],
   743→        });
   744→      }
   745→    }
   746→  }

   750→  private drawSelectionBox(ctx: CanvasRenderingContext2D, game: Game) {
   751→    const inp = game.input;
   752→    if (!inp.dragSelect || inp.cursorMode !== 'NORMAL') return;
   753→    const x = Math.min(inp.mouseDownX, inp.mouseX);
   754→    const y = Math.min(inp.mouseDownY, inp.mouseY);
   755→    const w = Math.abs(inp.mouseX - inp.mouseDownX);
   756→    const h = Math.abs(inp.mouseY - inp.mouseDownY);
   757→    if (w + h < 8) return;
   758→    ctx.fillStyle = 'rgba(20,18,15,0.05)';
   759→    ctx.fillRect(x, y, w, h);
   760→    ctx.strokeStyle = '#141210';
   761→    ctx.lineWidth = 1;
   762→    ctx.setLineDash([5, 4]);
   763→    ctx.strokeRect(x, y, w, h);
   764→    ctx.setLineDash([]);
   765→    // corner ticks
   766→    ctx.setLineDash([]);
   767→    ctx.lineWidth = 2;
   768→    const t = 7;
   769→    ctx.beginPath();
   770→    ctx.moveTo(x, y + t);
   771→    ctx.lineTo(x, y);
   772→    ctx.lineTo(x + t, y);
   773→    ctx.moveTo(x + w - t, y);
   774→    ctx.lineTo(x + w, y);
   775→    ctx.lineTo(x + w, y + t);
   776→    ctx.moveTo(x, y + h - t);
   777→    ctx.lineTo(x, y + h);
   778→    ctx.lineTo(x + t, y + h);
   779→    ctx.moveTo(x + w - t, y + h);
   780→    ctx.lineTo(x + w, y + h);
   781→    ctx.lineTo(x + w, y + h - t);
   782→    ctx.stroke();
   783→  }

   821→  drawMinimap(ctx: CanvasRenderingContext2D, w: number, h: number, game: Game) {
   822→    if (!this.minimapBase) {
   823→      this.minimapBase = document.createElement('canvas');
   824→      this.minimapBase.width = w;
   825→      this.minimapBase.height = h;
   826→      game.terrainRenderer.renderMinimapBase(this.minimapBase.getContext('2d')!, w, h);
   827→    }
   828→    ctx.clearRect(0, 0, w, h);
   829→    ctx.drawImage(this.minimapBase, 0, 0);
   830→
   831→    const sx = w / game.terrain.W;
   832→    const sy = h / game.terrain.H;
   833→
   834→    // unexplored fog
   835→    const vis = game.vision;
   836→    const cellW = w / vis.cols;
   837→    const cellH = h / vis.rows;
   838→    ctx.fillStyle = 'rgba(30,27,22,0.34)';
   839→    for (let gy = 0; gy < vis.rows; gy++) {
   840→      for (let gx = 0; gx < vis.cols; gx++) {
   841→        if (!vis.explored[gy * vis.cols + gx]) {
   842→          ctx.fillRect(gx * cellW, gy * cellH, cellW + 0.5, cellH + 0.5);
   843→        }
   844→      }
   845→    }
   846→
   847→    // sectors — ownership diamonds
   848→    for (const s of game.economy.sectors) {
   849→      const x = s.pos.x * sx;
   850→      const y = s.pos.y * sy;
   851→      ctx.save();
   852→      ctx.translate(x, y);
   853→      ctx.rotate(Math.PI / 4);
   854→      const s2 = 3.2;
   855→      if (s.control === 'FRIEND') {
   856→        ctx.fillStyle = '#0c0b08';
   857→        ctx.fillRect(-s2, -s2, s2 * 2, s2 * 2);
   858→      } else if (s.control === 'ENEMY') {
   859→        ctx.strokeStyle = 'rgba(110,104,92,0.95)';
   860→        ctx.lineWidth = 1;
   861→        ctx.strokeRect(-s2, -s2, s2 * 2, s2 * 2);
   862→      } else {
   863→        ctx.strokeStyle = 'rgba(130,124,110,0.8)';
   864→        ctx.lineWidth = 1;
   865→        ctx.beginPath();
   866→        ctx.arc(0, 0, s2, 0, Math.PI * 2);
   867→        ctx.stroke();
   868→      }
   869→      ctx.restore();
   870→    }
   871→
   872→    // objectives
   873→    for (const obj of game.objectives) {
   874→      ctx.strokeStyle = obj.secured ? '#141210' : 'rgba(30,27,22,0.85)';
   875→      ctx.lineWidth = 1;
   876→      const s = 5;
   877→      ctx.strokeRect(obj.pos.x * sx - s, obj.pos.y * sy - s, s * 2, s * 2);
   878→    }
   879→
   880→    // ink works — squares, filled by holder
   881→    for (const u of game.units) {
   882→      if (u.def.kind !== 'FACTORY') continue;
   883→      const x = u.x * sx;
   884→      const y = u.y * sy;
   885→      const s = 4;
   886→      if (u.dead) {
   887→        ctx.strokeStyle = 'rgba(20,17,12,0.8)';
   888→        ctx.lineWidth = 1;
   889→        ctx.beginPath();
   890→        ctx.moveTo(x - s, y - s);
   891→        ctx.lineTo(x + s, y + s);
   892→        ctx.moveTo(x + s, y - s);
   893→        ctx.lineTo(x - s, y + s);
   894→        ctx.stroke();
   895→      } else if (u.factoryCtl === 'FRIEND') {
   896→        ctx.fillStyle = '#0c0b08';
   897→        ctx.fillRect(x - s, y - s, s * 2, s * 2);
   898→      } else if (u.factoryCtl === 'NEUTRAL') {
   899→        ctx.strokeStyle = 'rgba(60,55,46,0.9)';
   900→        ctx.lineWidth = 1;
   901→        ctx.strokeRect(x - s, y - s, s * 2, s * 2);
   902→      } else {
   903→        ctx.fillStyle = 'rgba(110,104,92,0.95)';
   904→        ctx.fillRect(x - s, y - s, s * 2, s * 2);
   905→      }
   906→    }
   907→
   908→    // wrecks
   909→    ctx.fillStyle = 'rgba(20,17,12,0.5)';
   910→    for (const wr of game.effects.wrecks) {
   911→      ctx.fillRect(wr.x * sx - 1, wr.y * sy - 1, 2, 2);
   912→    }
   913→
   914→    // units
   915→    for (const u of game.units) {
   916→      if (u.dead) continue;
   917→      if (u.faction === 'FRIEND') {
   918→        if (u.isAir && (u.airState === 'STANDBY' || u.airState === 'REARM')) continue;
   919→        if (u.def.kind === 'FACTORY') continue;
   920→        ctx.fillStyle = '#0c0b08';
   921→        if (u.isAir) {
   922→          ctx.beginPath();
   923→          ctx.moveTo(u.x * sx, u.y * sy - 3);
   924→          ctx.lineTo(u.x * sx + 3, u.y * sy + 2);
   925→          ctx.lineTo(u.x * sx - 3, u.y * sy + 2);
   926→          ctx.closePath();
   927→          ctx.fill();
   928→        } else {
   929→          const s = 2.5;
   930→          ctx.fillRect(u.x * sx - s / 2, u.y * sy - s / 2, s, s);
   931→        }
   932→      } else {
   933→        if (u.def.kind === 'FACTORY') continue;
   934→        if (u.intel === 'DETECTED') {
   935→          const s = 2.5;
   936→          ctx.fillStyle = '#6b655a';
   937→          ctx.fillRect(u.x * sx - s / 2, u.y * sy - s / 2, s, s);
   938→        } else if (u.intel === 'GHOST') {
   939→          ctx.strokeStyle = 'rgba(90,84,74,0.8)';
   940→          ctx.lineWidth = 1;
   941→          ctx.strokeRect(u.knownX * sx - 1.5, u.knownY * sy - 1.5, 3, 3);
   942→        }
   943→      }
   944→    }
   945→
   946→    // viewport
   947→    const cam = game.camera;
   948→    ctx.strokeStyle = '#0c0b08';
   949→    ctx.lineWidth = 1;
   950→    ctx.strokeRect(cam.viewX * sx, cam.viewY * sy, (cam.viewW / cam.zoom) * sx, (cam.viewH / cam.zoom) * sy);
   951→  }
