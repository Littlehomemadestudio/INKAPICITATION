   123→

   182→  constructor(seed: number) {
   183→    this.seed = seed >>> 0;
   184→    this.baseNoise = new Noise2D(this.seed);
   185→    this.detailNoise = new Noise2D((this.seed ^ 0x9e3779b9) >>> 0);
   186→    this.forestNoise = new Noise2D((this.seed ^ 0x51ed2701) >>> 0);
   187→    this.sea = new Sea(this.seed, this.W, this.H);
   188→    this.generate();
   189→  }
   190→
   191→  // ── height ─────────────────────────────────────────────────
   192→  //
   193→  // Hand-composed relief — the map is authored, not noise soup:
   194→  //   · ZAPAD RIDGE  — elongated NE–SW ridgeline in the west
   195→  //   · HILL 214     — dominant high ground, east-centre
   196→  //   · HILL 163     — detached knoll north of the river
   197→  //   · southern rolling rise screening the player's entry
   198→  //   · NE plateau carrying the enemy HQ
   199→  //   · river valley carve + dry tributary
   200→  //
   201→  heightAt(x: number, y: number): number {
   202→    let h = this.baseNoise.fbm(x / 1500, y / 1500, 5) * 26 - 6;
   203→    h += this.detailNoise.fbm(x / 380, y / 380, 3) * 5 - 1.5;
   204→
   205→    // ZAPAD RIDGE — four overlapping crests along one spine
   206→    h += this.gaussEl(x, y, 500, 480, 380, 300, 28);
   207→    h += this.gaussEl(x, y, 730, 660, 420, 320, 36);
   208→    h += this.gaussEl(x, y, 950, 840, 400, 300, 30);
   209→    h += this.gaussEl(x, y, 1140, 1010, 360, 280, 22);
   210→
   211→    // HILL 214 — the dominant landform
   212→    h += this.gauss(x, y, 2950, 1180, 620, 58);
   213→    // shoulder south-west of the hill (finger toward the village)
   214→    h += this.gaussEl(x, y, 2620, 1420, 360, 260, 14);
   215→
   216→    // HILL 163 knoll
   217→    h += this.gauss(x, y, 1550, 680, 360, 26);
   218→
   219→    // southern rise near player entry
   220→    h += this.gauss(x, y, 1750, 2750, 520, 16);
   221→    // south-eastern rolling ground
   222→    h += this.gaussEl(x, y, 3350, 2600, 600, 380, 12);
   223→
   224→    // north-east plateau toward HQ
   225→    const nx = clamp(x / this.W, 0, 1);
   226→    const ny = 1 - clamp(y / this.H, 0, 1);
   227→    h += smooth01(nx * 0.7 + ny * 0.7) * 22;
   228→
   229→    // river valley carve
   230→    const dRiver = this.distToPolyline(x, y, this.river);
   231→    if (dRiver < 170) {
   232→      const t = 1 - dRiver / 170;
   233→      h -= smooth01(t) * 11;
   234→    }
   235→    // islands rise out of the bay — real ground for LOS and contours
   236→    for (const isl of this.sea.islands) {
   237→      h += this.gaussEl(x, y, isl.x, isl.y, isl.rx * 1.35, isl.ry * 1.35, isl.height);
   238→    }
   239→    // the sea floor is flat: the water surface carries the depth story
   240→    if (this.sea.isSea(x, y)) return 0;
   241→    return Math.max(0, h);
   242→  }
   243→
   244→  slopeAt(x: number, y: number): number {
   245→    const d = 24;
   246→    const hx = this.heightAt(x + d, y) - this.heightAt(x - d, y);
   247→    const hy = this.heightAt(x, y + d) - this.heightAt(x, y - d);
   248→    return Math.hypot(hx, hy) / (2 * d);
   249→  }
   250→
   251→  private gauss(x: number, y: number, cx: number, cy: number, r: number, amp: number): number {
   252→    const d = dist(x, y, cx, cy);
   253→    if (d > r) return 0;
   254→    const t = 1 - d / r;
   255→    return amp * t * t * (3 - 2 * t);
   256→  }
   257→
   258→  /** elongated gaussian — ridge segments */
   259→  private gaussEl(x: number, y: number, cx: number, cy: number, rx: number, ry: number, amp: number): number {
   260→    const dx = (x - cx) / rx;
   261→    const dy = (y - cy) / ry;
   262→    const d2 = dx * dx + dy * dy;
   263→    if (d2 > 1) return 0;
   264→    const t = 1 - Math.sqrt(d2);
   265→    return amp * t * t * (3 - 2 * t);
   266→  }

   268→  // ── LOS over the height grid ───────────────────────────────
   269→
   270→  /** terrain line of sight between two points at given eye heights;
   271→   *  buildings interrupt observation and direct fire */
   272→  losClear(ax: number, ay: number, aEye: number, bx: number, by: number, bEye: number, includeBuildings = true): boolean {
   273→    const d = dist(ax, ay, bx, by);
   274→    const steps = Math.max(2, Math.ceil(d / 56));
   275→    for (let s = 1; s < steps; s++) {
   276→      const t = s / steps;
   277→      const x = ax + (bx - ax) * t;
   278→      const y = ay + (by - ay) * t;
   279→      const beam = aEye + (bEye - aEye) * t;
   280→      const ground = this.gridHeight(x, y);
   281→      if (ground > beam + 2.0) return false;
   282→    }
   283→    if (includeBuildings) {
   284→      // sample the beam against building footprints (they are tall enough
   285→      // to matter to a ground-level observer; aircraft pass above)
   286→      if (aEye < 30 && bEye < 30) {
   287→        const bsteps = Math.max(2, Math.ceil(d / 16));
   288→        for (let s = 1; s < bsteps; s++) {
   289→          const t = s / bsteps;
   290→          const x = ax + (bx - ax) * t;
   291→          const y = ay + (by - ay) * t;
   292→          const gx = (x / this.BM_CELL) | 0;
   293→          const gy = (y / this.BM_CELL) | 0;
   294→          if (gx < 0 || gy < 0 || gx >= this.bmw || gy >= this.bmh) continue;
   295→          const idx = gy * this.bmw + gx;
   296→          if (this.bmask[idx] || this.rockMask[idx]) return false;
   297→        }
   298→      }
   299→    }
   300→    return true;
   301→  }
   302→
   303→  private gridHeight(x: number, y: number): number {
   304→    const gx = clamp(x / HG_STEP, 0, this.hgw - 1.001);
   305→    const gy = clamp(y / HG_STEP, 0, this.hgh - 1.001);
   306→    const ix = Math.floor(gx);
   307→    const iy = Math.floor(gy);
   308→    const fx = gx - ix;
   309→    const fy = gy - iy;
   310→    const h00 = this.hg[iy * this.hgw + ix];
   311→    const h10 = this.hg[iy * this.hgw + ix + 1];
   312→    const h01 = this.hg[(iy + 1) * this.hgw + ix];
   313→    const h11 = this.hg[(iy + 1) * this.hgw + ix + 1];
   314→    return (h00 * (1 - fx) + h10 * fx) * (1 - fy) + (h01 * (1 - fx) + h11 * fx) * fy;
   315→  }

   335→  /** shallow ford — vehicles may wade across, slowly */
   336→  fordAt(x: number, y: number, r = 46): boolean {
   337→    if (!this.ford) return false;
   338→    return dist(x, y, this.ford.x, this.ford.y) < r;
   339→  }

   341→  /** bridge within radius */
   342→  bridgeAt(x: number, y: number, r = 42): Bridge | null {
   343→    for (const b of this.bridges) {
   344→      if (dist(x, y, b.x, b.y) < r) return b;
   345→    }
   346→    return null;
   347→  }

   349→  // ── roads & forests ────────────────────────────────────────

   351→  roadFactor(x: number, y: number): number {
   352→    // 0 = no road, 1 = on major road centreline (approx)
   353→    let best = 0;
   354→    for (const r of this.roads) {
   355→      const d = this.distToPolyline(x, y, r.pts);
   356→      const w = r.major ? 26 : 20;
   357→      if (d < w) best = Math.max(best, (r.major ? 1 : 0.7) * (1 - d / w * 0.4));
   358→    }
   359→    return best;
   360→  }
   361→
   362→  railFactor(x: number, y: number): number {
   363→    const d = this.distToPolyline(x, y, this.railway);
   364→    return d < 14 ? 1 - d / 14 * 0.5 : 0;
   365→  }
   366→
   367→  forestDensity(x: number, y: number): number {
   368→    const n = this.forestNoise.fbm(x / 520, y / 520, 4);
   369→    // NW-biased woodland belt; the valley slopes carry brush
   370→    const west = 1 - x / this.W;
   371→    const bias = -0.075 * west + 0.035 * (y / this.H);
   372→    const v = (n - 0.585 + bias) * 4.2;
   373→    return clamp(v, 0, 1) * 1.6;
   374→  }

   878→  /** true if a w×h footprint at (x,y,rot) keeps `margin` m clear of every standing building */
   879→  private buildingFits(x: number, y: number, w: number, h: number, rot: number, margin = 4): boolean {
   880→    const reach = Math.hypot(w, h) / 2;
   881→    for (const b of this.buildings) {
   882→      const br = Math.hypot(b.w, b.h) / 2;
   883→      if (Math.abs(b.x - x) > reach + br + margin || Math.abs(b.y - y) > reach + br + margin) continue;
   884→      // rotated-box SAT — minimum separation across the four axes
   885→      const ca1 = Math.cos(rot), sa1 = Math.sin(rot);
   886→      const ca2 = Math.cos(b.rot), sa2 = Math.sin(b.rot);
   887→      const dx = b.x - x, dy = b.y - y;
   888→      const axes: Array<[number, number]> = [[ca1, sa1], [-sa1, ca1], [ca2, sa2], [-sa2, ca2]];
   889→      let minSep = Infinity;
   890→      for (const [ux, uy] of axes) {
   891→        const dist = dx * ux + dy * uy;
   892→        const rA = (w / 2) * Math.abs(ca1 * ux + sa1 * uy) + (h / 2) * Math.abs(-sa1 * ux + ca1 * uy);
   893→        const rB = (b.w / 2) * Math.abs(ca2 * ux + sa2 * uy) + (b.h / 2) * Math.abs(-sa2 * ux + ca2 * uy);
   894→        minSep = Math.min(minSep, rA + rB - Math.abs(dist));
   895→      }
   896→      if (minSep > -margin) return false;
   897→    }
   898→    return true;
   899→  }

   901→  private buildTown(cx: number, cy: number, rng: RNG) {
   902→    // buildings string along both roads — a real crossroads town
   903→    const along = (
   904→      dx: number,
   905→      dy: number,
   906→      n: number,
   907→      side: number,
   908→      kinds: BuildingKind[]
   909→    ) => {
   910→      const span = 210; // metres of road frontage the lots share
   911→      for (let i = 0; i < n; i++) {
   912→        const t = (i + 1) / (n + 1);
   913→        const kind = kinds[i % kinds.length];
   914→        const w = kind === 'BARN' ? rng.range(26, 32) : rng.range(15, 22);
   915→        const h = kind === 'BARN' ? rng.range(14, 17) : rng.range(11, 15);
   916→        // try a few setbacks — a lot that doesn't fit is left empty
   917→        let placed = false;
   918→        for (let attempt = 0; attempt < 3 && !placed; attempt++) {
   919→          const set = 52 + attempt * 16 + rng.range(-6, 6);
   920→          const bx = cx + dx * (t - 0.5) * span + -dy * side * set;
   921→          const by = cy + dy * (t - 0.5) * span + dx * side * set;
   922→          if (this.isWater(bx, by) || this.roadFactor(bx, by) > 0.1) continue;
   923→          const rot = Math.atan2(dy, dx) + rng.range(-0.12, 0.12);
   924→          if (!this.buildingFits(bx, by, w, h, rot)) continue;
   925→          this.buildings.push({ x: bx, y: by, w, h, rot, kind });
   926→          placed = true;
   927→        }
   928→      }
   929→    };
   930→    along(1, -0.36, 4, 1, ['HOUSE', 'HOUSE', 'BARN', 'HOUSE']); // along MSR north
   931→    along(1, -0.36, 3, -1, ['HOUSE', 'SHED', 'HOUSE']);

  1074→  onBuildingDestroyed(b: Building) {
  1075→    // clear LOS occupancy over the footprint (intersect-based, matching
  1076→    // the rasterisation so no stale cells survive)
  1077→    const reach = Math.max(b.w, b.h) * 0.5;
  1078→    const gx0 = Math.max(0, ((b.x - reach) / this.BM_CELL) | 0);
  1079→    const gx1 = Math.min(this.bmw - 1, ((b.x + reach) / this.BM_CELL) | 0);
  1080→    const gy0 = Math.max(0, ((b.y - reach) / this.BM_CELL) | 0);
  1081→    const gy1 = Math.min(this.bmh - 1, ((b.y + reach) / this.BM_CELL) | 0);

  1083→    const s = Math.sin(b.rot);
  1084→    for (let gy = gy0; gy <= gy1; gy++) {
  1085→      for (let gx = gx0; gx <= gx1; gx++) {
  1086→        const px = clamp(b.x, gx * this.BM_CELL, (gx + 1) * this.BM_CELL);
  1087→        const py = clamp(b.y, gy * this.BM_CELL, (gy + 1) * this.BM_CELL);
  1088→        const dx = px - b.x;
  1089→        const dy = py - b.y;
  1090→        const lx = dx * c + dy * s;
  1091→        const ly = -dx * s + dy * c;
  1092→        if (Math.abs(lx) <= b.w / 2 && Math.abs(ly) <= b.h / 2) {
  1093→          this.bmask[gy * this.bmw + gx] = 0;
  1094→        }
  1095→      }
  1096→    }
  1097→    // rubble is passable — route cost eases where the structure stood
  1098→    const c0x = Math.max(0, ((b.x - reach) / CELL) | 0);
  1099→    const c1x = Math.min(this.gw - 1, ((b.x + reach) / CELL) | 0);
  1100→    const c0y = Math.max(0, ((b.y - reach) / CELL) | 0);
  1101→    const c1y = Math.min(this.gh - 1, ((b.y + reach) / CELL) | 0);
  1102→    for (let gy = c0y; gy <= c1y; gy++) {
  1103→      for (let gx = c0x; gx <= c1x; gx++) {
  1104→        const i = gy * this.gw + gx;
  1105→        if (this.cost[i] > 0) this.cost[i] = Math.max(0.9, this.cost[i] - 5);
  1106→      }
  1107→    }
  1108→  }

  1111→  wallSegmentAlive(w: StoneWall, x: number, y: number): boolean {
  1112→    if (!w.segs || !w.segs.length) return true;
  1113→    const c = Math.cos(w.rot);
  1114→    const s = Math.sin(w.rot);
  1115→    const dx = x - w.x;

  1123→  }

  1167→  private extractContours(): ContourSet {
  1168→    const step = 16; // metres between samples
  1169→    const nx = Math.floor(this.W / step) + 1;
  1170→    const ny = Math.floor(this.H / step) + 1;
  1171→    const heights = new Float32Array(nx * ny);
  1172→    for (let j = 0; j < ny; j++) {
  1173→      for (let i = 0; i < nx; i++) {
  1174→        heights[j * nx + i] = this.heightAt(i * step, j * step);
  1175→      }
  1176→    }
  1177→    const minor = new Path2D();
  1178→    const major = new Path2D();
  1179→    const interval = 8;
  1180→    const majorEvery = 5;
  1181→
  1182→    const lerpPt = (x1: number, y1: number, x2: number, y2: number, h1: number, h2: number, level: number) => {
  1183→      const t = (level - h1) / (h2 - h1 || 1e-6);
  1184→      return { x: (x1 + (x2 - x1) * t) * step, y: (y1 + (y2 - y1) * t) * step };
  1185→    };
  1186→
  1187→    for (let j = 0; j < ny - 1; j++) {
  1188→      for (let i = 0; i < nx - 1; i++) {
  1189→        const x = i * step;
  1190→        const y = j * step;
  1191→        const h00 = heights[j * nx + i];
  1192→        const h10 = heights[j * nx + i + 1];
  1193→        const h01 = heights[(j + 1) * nx + i];
  1194→        const h11 = heights[(j + 1) * nx + i + 1];
  1195→        const hMin = Math.min(h00, h10, h01, h11);
  1196→        const hMax = Math.max(h00, h10, h01, h11);
  1197→        const firstLevel = Math.ceil(hMin / interval) * interval;
  1198→        for (let level = firstLevel; level <= hMax; level += interval) {
  1199→          if (level <= 0) continue;
  1200→          const isMajor = Math.round(level / interval) % majorEvery === 0;
  1201→          const path = isMajor ? major : minor;
  1202→          // marching squares on cell corners
  1203→          let idx = 0;
  1204→          if (h00 >= level) idx |= 1;
  1205→          if (h10 >= level) idx |= 2;
  1206→          if (h11 >= level) idx |= 4;
  1207→          if (h01 >= level) idx |= 8;
  1208→          if (idx === 0 || idx === 15) continue;
  1209→          const top = () => lerpPt(i, j, i + 1, j, h00, h10, level);
  1210→          const right = () => lerpPt(i + 1, j, i + 1, j + 1, h10, h11, level);
  1211→          const bottom = () => lerpPt(i, j + 1, i + 1, j + 1, h01, h11, level);
  1212→          const left = () => lerpPt(i, j, i, j + 1, h00, h01, level);
  1213→          const seg = (a: { x: number; y: number }, b: { x: number; y: number }) => {
  1214→            path.moveTo(a.x, a.y);
  1215→            path.lineTo(b.x, b.y);
  1216→          };
  1217→          switch (idx) {
  1218→            case 1:
  1219→            case 14:
  1220→              seg(left(), top());
  1221→              break;
  1222→            case 2:
  1223→            case 13:
  1224→              seg(top(), right());
  1225→              break;
  1226→            case 3:
  1227→            case 12:
  1228→              seg(left(), right());
  1229→              break;
  1230→            case 4:
  1231→            case 11:
  1232→              seg(right(), bottom());
  1233→              break;
  1234→            case 6:
  1235→            case 9:
  1236→              seg(top(), bottom());
  1237→              break;
  1238→            case 7:
  1239→            case 8:
  1240→              seg(left(), bottom());
  1241→              break;
  1242→            case 5:
  1243→              seg(left(), top());
  1244→              seg(right(), bottom());
  1245→              break;
  1246→            case 10:
  1247→              seg(top(), right());
  1248→              seg(left(), bottom());
  1249→              break;
  1250→            default:
  1251→              break;
  1252→          }
  1253→        }
  1254→      }
  1255→    }
  1256→    return { minor, major };
  1257→  }

  1261→  private buildCostGrid() {
  1262→    this.gw = Math.ceil(this.W / CELL);
  1263→    this.gh = Math.ceil(this.H / CELL);
  1264→    const n = this.gw * this.gh;
  1265→    this.cost = new Float32Array(n);
  1266→    for (let gy = 0; gy < this.gh; gy++) {
  1267→      for (let gx = 0; gx < this.gw; gx++) {
  1268→        const cx = gx * CELL + CELL / 2;
  1269→        const cy = gy * CELL + CELL / 2;
  1270→        let c: number;
  1271→        if (this.isWater(cx, cy) && !this.bridgeAt(cx, cy, CELL * 0.9) && !this.fordAt(cx, cy, CELL)) {
  1272→          c = -1; // impassable water
  1273→        } else if (this.fordAt(cx, cy, CELL)) {
  1274→          c = 3.6; // wading — slow but possible
  1275→        } else {
  1276→          c = 1;
  1277→          c += this.slopeAt(cx, cy) * 14;
  1278→          const f = this.forestDensity(cx, cy);
  1279→          if (f > 0.42) c += 2.4;
  1280→          const r = this.roadFactor(cx, cy);
  1281→          if (r > 0.05) c = Math.min(c, r > 0.5 ? 0.34 : 0.6);
  1282→          if (this.railFactor(cx, cy) > 0.4) c += 1.2;
  1283→          if (this.buildingAt(cx, cy, 10)) c += 5;
  1284→          if (this.trenchDist(cx, cy) < 8) c += 1.4; // crossing a dug position
  1285→          // boulder fields and concrete are avoided by preference
  1286→          for (const rk of this.rocks) {
  1287→            if (rk.r < 2.6) continue;
  1288→            if (dist(cx, cy, rk.x, rk.y) < rk.r + 18) {
  1289→              c += 2.2;
  1290→              break;
  1291→            }
  1292→          }
  1293→          for (const b of this.barriers) {
  1294→            if ((b.hp ?? 100) <= 0) continue;
  1295→            if (dist(cx, cy, b.x, b.y) < 26) {
  1296→              c += 3.2;
  1297→              break;
  1298→            }
  1299→          }
  1300→          if (this.bridgeAt(cx, cy, CELL * 0.9)) c = 0.5;
  1301→        }
  1302→        this.cost[gy * this.gw + gx] = c;
  1303→      }
  1304→    }
  1305→  }

  1307→  private buildHeightGrid() {
  1308→    this.hgw = Math.ceil(this.W / HG_STEP) + 1;
  1309→    this.hgh = Math.ceil(this.H / HG_STEP) + 1;
  1310→    this.hg = new Float32Array(this.hgw * this.hgh);
  1311→    for (let j = 0; j < this.hgh; j++) {
  1312→      for (let i = 0; i < this.hgw; i++) {
  1313→        this.hg[j * this.hgw + i] = this.heightAt(i * HG_STEP, j * HG_STEP);
  1314→      }
  1315→    }
  1316→    // building occupancy mask — rasterize every footprint once
  1317→    this.bmw = Math.ceil(this.W / this.BM_CELL) + 1;
  1318→    this.bmh = Math.ceil(this.H / this.BM_CELL) + 1;
  1319→    this.bmask = new Uint8Array(this.bmw * this.bmh);
  1320→    // big boulders interrupt ground-level sightlines too
  1321→    this.rockMask = new Uint8Array(this.bmw * this.bmh);
  1322→    for (const rk of this.rocks) {
  1323→      if (rk.r < this.ROCK_LOS_R) continue;
  1324→      const g0x = Math.max(0, ((rk.x - rk.r) / this.BM_CELL) | 0);
  1325→      const g1x = Math.min(this.bmw - 1, ((rk.x + rk.r) / this.BM_CELL) | 0);
  1326→      const g0y = Math.max(0, ((rk.y - rk.r) / this.BM_CELL) | 0);
  1327→      const g1y = Math.min(this.bmh - 1, ((rk.y + rk.r) / this.BM_CELL) | 0);
  1328→      for (let gy = g0y; gy <= g1y; gy++) {
  1329→        for (let gx = g0x; gx <= g1x; gx++) {
  1330→          const px = gx * this.BM_CELL + this.BM_CELL / 2;
  1331→          const py = gy * this.BM_CELL + this.BM_CELL / 2;
  1332→          if (dist(px, py, rk.x, rk.y) < rk.r * 0.9) this.rockMask[gy * this.bmw + gx] = 1;
  1333→        }
  1334→      }
  1335→    }
  1336→    for (const b of this.buildings) {
  1337→      if (b.kind === 'MAST' || b.kind === 'CHECKPOINT') continue; // see-through
  1338→      const c = Math.cos(b.rot);
  1339→      const s = Math.sin(b.rot);
  1340→      // intersect-based rasterisation: any cell the footprint touches is
  1341→      // masked, so even a small house reliably interrupts a ground beam
  1342→      const reach = Math.max(b.w, b.h) * 0.5;
  1343→      const gx0 = Math.max(0, ((b.x - reach) / this.BM_CELL) | 0);
  1344→      const gx1 = Math.min(this.bmw - 1, ((b.x + reach) / this.BM_CELL) | 0);
  1345→      const gy0 = Math.max(0, ((b.y - reach) / this.BM_CELL) | 0);
  1346→      const gy1 = Math.min(this.bmh - 1, ((b.y + reach) / this.BM_CELL) | 0);
  1347→      for (let gy = gy0; gy <= gy1; gy++) {
  1348→        for (let gx = gx0; gx <= gx1; gx++) {
  1349→          // closest point of the cell rect to the building centre, in
  1350→          // building-local space — if inside the half-extents, mask
  1351→          const px = clamp(b.x, gx * this.BM_CELL, (gx + 1) * this.BM_CELL);
  1352→          const py = clamp(b.y, gy * this.BM_CELL, (gy + 1) * this.BM_CELL);
  1353→          const dx = px - b.x;
  1354→          const dy = py - b.y;
  1355→          const lx = dx * c + dy * s;
  1356→          const ly = -dx * s + dy * c;
  1357→          if (Math.abs(lx) <= b.w / 2 && Math.abs(ly) <= b.h / 2) {
  1358→            this.bmask[gy * this.bmw + gx] = 1;
  1359→          }
  1360→        }
  1361→      }
  1362→    }
  1363→  }
  1364→
  1365→  findPath(from: Vec2, to: Vec2): Vec2[] {
  1366→    const sgx = clamp(Math.floor(from.x / CELL), 0, this.gw - 1);
  1367→    const sgy = clamp(Math.floor(from.y / CELL), 0, this.gh - 1);
  1368→    let tgx = clamp(Math.floor(to.x / CELL), 0, this.gw - 1);
  1369→    let tgy = clamp(Math.floor(to.y / CELL), 0, this.gh - 1);
  1370→    if (this.cost[tgy * this.gw + tgx] < 0) {
  1371→      // nudge target to nearest passable cell
  1372→      const alt = this.nearestPassable(tgx, tgy);
  1373→      if (!alt) return [{ ...to }];
  1374→      tgx = alt[0];
  1375→      tgy = alt[1];
  1376→    }
  1377→    if (sgx === tgx && sgy === tgy) return [{ ...to }];
  1378→
  1379→    const n = this.gw * this.gh;
  1380→    const gScore = new Float32Array(n).fill(Infinity);
  1381→    const cameFrom = new Int32Array(n).fill(-1);
  1382→    const closed = new Uint8Array(n);
  1383→    const open: number[] = [];
  1384→    const fScore = new Float32Array(n).fill(Infinity);
  1385→    const h = (x: number, y: number) => (Math.abs(x - tgx) + Math.abs(y - tgy)) * 0.6;
  1386→    const start = sgy * this.gw + sgx;
  1387→    gScore[start] = 0;
  1388→    fScore[start] = h(sgx, sgy);
  1389→    open.push(start);
  1390→    let guard = 0;
  1391→    while (open.length && guard++ < 12000) {
  1392→      // find best in open
  1393→      let bi = 0;
  1394→      for (let i = 1; i < open.length; i++) {
  1395→        if (fScore[open[i]] < fScore[open[bi]]) bi = i;
  1396→      }
  1397→      const cur = open.splice(bi, 1)[0];
  1398→      if (cur === tgy * this.gw + tgx) break;
  1399→      closed[cur] = 1;
  1400→      const cx = cur % this.gw;
  1401→      const cy = (cur / this.gw) | 0;
  1402→      for (let dy = -1; dy <= 1; dy++) {
  1403→        for (let dx = -1; dx <= 1; dx++) {
  1404→          if (!dx && !dy) continue;
  1405→          const nx = cx + dx;
  1406→          const ny = cy + dy;
  1407→          if (nx < 0 || ny < 0 || nx >= this.gw || ny >= this.gh) continue;
  1408→          const ni = ny * this.gw + nx;
  1409→          if (closed[ni] || this.cost[ni] < 0) continue;
  1410→          const step = (dx && dy ? 1.414 : 1) * (this.cost[ni] + this.cost[cur]) * 0.5;
  1411→          const tentative = gScore[cur] + step;
  1412→          if (tentative < gScore[ni]) {
  1413→            gScore[ni] = tentative;
  1414→            cameFrom[ni] = cur;
  1415→            fScore[ni] = tentative + h(nx, ny);
  1416→            if (!open.includes(ni)) open.push(ni);
  1417→          }
  1418→        }
  1419→      }
  1420→    }
  1421→
  1422→    const end = tgy * this.gw + tgx;
  1423→    if (cameFrom[end] === -1 && end !== start) return [{ ...to }];
  1424→    // reconstruct
  1425→    const cells: number[] = [];
  1426→    let c = end;
  1427→    while (c !== -1 && c !== start) {
  1428→      cells.push(c);
  1429→      c = cameFrom[c];
  1430→    }
  1431→    cells.push(start);
  1432→    cells.reverse();
  1433→    // waypoints at cell centres, then string-pull
  1434→    let pts: Vec2[] = cells.map((ci) => ({
  1435→      x: (ci % this.gw) * CELL + CELL / 2,
  1436→      y: ((ci / this.gw) | 0) * CELL + CELL / 2,
  1437→    }));
  1438→    pts.push({ ...to });
  1439→    pts = this.smoothPath(pts);
  1440→    return pts.length ? pts : [{ ...to }];
  1441→  }
  1442→
  1443→  private nearestPassable(gx: number, gy: number): [number, number] | null {
  1444→    for (let r = 1; r < 8; r++) {
  1445→      for (let dy = -r; dy <= r; dy++) {
  1446→        for (let dx = -r; dx <= r; dx++) {
  1447→          const nx = gx + dx;
  1448→          const ny = gy + dy;
  1449→          if (nx < 0 || ny < 0 || nx >= this.gw || ny >= this.gh) continue;
  1450→          if (this.cost[ny * this.gw + nx] >= 0) return [nx, ny];
  1451→        }
  1452→      }
  1453→    }
  1454→    return null;
  1455→  }
  1456→
  1457→  private smoothPath(pts: Vec2[]): Vec2[] {
  1458→    if (pts.length < 3) return pts;
  1459→    const out: Vec2[] = [pts[0]];
  1460→    let i = 0;
  1461→    while (i < pts.length - 1) {
  1462→      let j = pts.length - 1;
  1463→      for (; j > i + 1; j--) {
  1464→        if (this.los(pts[i], pts[j])) break;
  1465→      }
  1466→      out.push(pts[j]);
  1467→      i = j;
  1468→    }
  1469→    return out;
  1470→  }

  1472→  los(a: Vec2, b: Vec2): boolean {
  1473→    const d = dist(a.x, a.y, b.x, b.y);
  1474→    const steps = Math.max(2, Math.ceil(d / (CELL * 0.6)));
  1475→    for (let s = 1; s < steps; s++) {
  1476→      const t = s / steps;
  1477→      const x = a.x + (b.x - a.x) * t;
  1478→      const y = a.y + (b.y - a.y) * t;
  1479→      if (this.isWater(x, y) && !this.bridgeAt(x, y, 30) && !this.fordAt(x, y, 40)) return false;
  1480→      if (this.slopeAt(x, y) > 0.75) return false;
  1481→    }
  1482→    return true;
  1483→  }

  1506→  distToPolyline(x: number, y: number, pts: Vec2[]): number {
  1507→    let best = Infinity;
  1508→    for (let i = 0; i < pts.length - 1; i++) {
  1509→      const a = pts[i];
  1510→      const b = pts[i + 1];
  1511→      const vx = b.x - a.x;
  1512→      const vy = b.y - a.y;
  1513→      const wx = x - a.x;
  1514→      const wy = y - a.y;
  1515→      const len2 = vx * vx + vy * vy || 1e-6;
  1516→      let t = (wx * vx + wy * vy) / len2;
  1517→      t = clamp(t, 0, 1);
  1518→      const dx = x - (a.x + vx * t);
  1519→      const dy = y - (a.y + vy * t);
  1520→      const d = dx * dx + dy * dy;
  1521→      if (d < best) best = d;
  1522→    }
  1523→    return Math.sqrt(best);
  1524→  }

  1526→  polylineDirAt(x: number, y: number, pts: Vec2[]): number {
  1527→    let best = Infinity;
  1528→    let bestI = 0;
  1529→    let bestT = 0;
  1530→    for (let i = 0; i < pts.length - 1; i++) {
  1531→      const a = pts[i];
  1532→      const b = pts[i + 1];
  1533→      const vx = b.x - a.x;
  1534→      const vy = b.y - a.y;
  1535→      const wx = x - a.x;
  1536→      const wy = y - a.y;
  1537→      const len2 = vx * vx + vy * vy || 1e-6;
  1538→      let t = (wx * vx + wy * vy) / len2;
  1539→      t = clamp(t, 0, 1);
  1540→      const dx = x - (a.x + vx * t);
  1541→      const dy = y - (a.y + vy * t);
  1542→      const d = dx * dx + dy * dy;
  1543→      if (d < best) {
  1544→        best = d;
  1545→        bestI = i;
  1546→        bestT = t;
  1547→      }
  1548→    }
  1549→    const a = pts[bestI];
  1550→    const b = pts[bestI + 1];
  1551→    void bestT;
  1552→    return Math.atan2(b.y - a.y, b.x - a.x);
  1553→  }
