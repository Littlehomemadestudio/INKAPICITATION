   160→  constructor(seed: number) {
   161→    this.seed = seed >>> 0;
   162→    this.baseNoise = new Noise2D(this.seed);
   163→    this.detailNoise = new Noise2D((this.seed ^ 0x9e3779b9) >>> 0);

   168→  // ── height ─────────────────────────────────────────────────
   169→  //
   170→  // Hand-composed relief — the map is authored, not noise soup:
   171→  //   · ZAPAD RIDGE  — elongated NE–SW ridgeline in the west
   172→  //   · HILL 214     — dominant high ground, east-centre
   173→  //   · HILL 163     — detached knoll north of the river
   174→  //   · southern rolling rise screening the player's entry
   175→  //   · NE plateau carrying the enemy HQ
   176→  //   · river valley carve + dry tributary
   177→  //
   178→  heightAt(x: number, y: number): number {
   179→    let h = this.baseNoise.fbm(x / 1500, y / 1500, 5) * 26 - 6;
   180→    h += this.detailNoise.fbm(x / 380, y / 380, 3) * 5 - 1.5;
   181→
   182→    // ZAPAD RIDGE — four overlapping crests along one spine
   183→    h += this.gaussEl(x, y, 500, 480, 380, 300, 28);
   184→    h += this.gaussEl(x, y, 730, 660, 420, 320, 36);
   185→    h += this.gaussEl(x, y, 950, 840, 400, 300, 30);
   186→    h += this.gaussEl(x, y, 1140, 1010, 360, 280, 22);
   187→
   188→    // HILL 214 — the dominant landform
   189→    h += this.gauss(x, y, 2950, 1180, 620, 58);
   190→    // shoulder south-west of the hill (finger toward the village)
   191→    h += this.gaussEl(x, y, 2620, 1420, 360, 260, 14);
   192→
   193→    // HILL 163 knoll
   194→    h += this.gauss(x, y, 1550, 680, 360, 26);
   195→
   196→    // southern rise near player entry
   197→    h += this.gauss(x, y, 1750, 2750, 520, 16);
   198→    // south-eastern rolling ground
   199→    h += this.gaussEl(x, y, 3350, 2600, 600, 380, 12);
   200→
   201→    // north-east plateau toward HQ
   202→    const nx = clamp(x / this.W, 0, 1);
   203→    const ny = 1 - clamp(y / this.H, 0, 1);
   204→    h += smooth01(nx * 0.7 + ny * 0.7) * 22;
   205→
   206→    // river valley carve
   207→    const dRiver = this.distToPolyline(x, y, this.river);
   208→    if (dRiver < 170) {
   209→      const t = 1 - dRiver / 170;
   210→      h -= smooth01(t) * 11;
   211→    }
   212→    return Math.max(0, h);
   213→  }
   214→
   215→  slopeAt(x: number, y: number): number {
   216→    const d = 24;
   217→    const hx = this.heightAt(x + d, y) - this.heightAt(x - d, y);
   218→    const hy = this.heightAt(x, y + d) - this.heightAt(x, y - d);
   219→    return Math.hypot(hx, hy) / (2 * d);
   220→  }
   221→
   222→  private gauss(x: number, y: number, cx: number, cy: number, r: number, amp: number): number {
   223→    const d = dist(x, y, cx, cy);
   224→    if (d > r) return 0;
   225→    const t = 1 - d / r;
   226→    return amp * t * t * (3 - 2 * t);
   227→  }
   228→
   229→  /** elongated gaussian — ridge segments */
   230→  private gaussEl(x: number, y: number, cx: number, cy: number, rx: number, ry: number, amp: number): number {
   231→    const dx = (x - cx) / rx;
   232→    const dy = (y - cy) / ry;
   233→    const d2 = dx * dx + dy * dy;
   234→    if (d2 > 1) return 0;
   235→    const t = 1 - Math.sqrt(d2);
   236→    return amp * t * t * (3 - 2 * t);
   237→  }

   239→  // ── LOS over the height grid ───────────────────────────────
   240→
   241→  /** terrain line of sight between two points at given eye heights;
   242→   *  buildings interrupt observation and direct fire */
   243→  losClear(ax: number, ay: number, aEye: number, bx: number, by: number, bEye: number, includeBuildings = true): boolean {
   244→    const d = dist(ax, ay, bx, by);
   245→    const steps = Math.max(2, Math.ceil(d / 56));
   246→    for (let s = 1; s < steps; s++) {
   247→      const t = s / steps;
   248→      const x = ax + (bx - ax) * t;
   249→      const y = ay + (by - ay) * t;
   250→      const beam = aEye + (bEye - aEye) * t;
   251→      const ground = this.gridHeight(x, y);
   252→      if (ground > beam + 2.0) return false;
   253→    }
   254→    if (includeBuildings) {
   255→      // sample the beam against building footprints (they are tall enough
   256→      // to matter to a ground-level observer; aircraft pass above)
   257→      if (aEye < 30 && bEye < 30) {
   258→        const bsteps = Math.max(2, Math.ceil(d / 16));
   259→        for (let s = 1; s < bsteps; s++) {
   260→          const t = s / bsteps;
   261→          const x = ax + (bx - ax) * t;
   262→          const y = ay + (by - ay) * t;
   263→          const gx = (x / this.BM_CELL) | 0;
   264→          const gy = (y / this.BM_CELL) | 0;
   265→          if (gx < 0 || gy < 0 || gx >= this.bmw || gy >= this.bmh) continue;
   266→          if (this.bmask[gy * this.bmw + gx]) return false;
   267→        }
   268→      }
   269→    }
   270→    return true;
   271→  }
   272→
   273→  private gridHeight(x: number, y: number): number {
   274→    const gx = clamp(x / HG_STEP, 0, this.hgw - 1.001);
   275→    const gy = clamp(y / HG_STEP, 0, this.hgh - 1.001);
   276→    const ix = Math.floor(gx);
   277→    const iy = Math.floor(gy);
   278→    const fx = gx - ix;
   279→    const fy = gy - iy;
   280→    const h00 = this.hg[iy * this.hgw + ix];
   281→    const h10 = this.hg[iy * this.hgw + ix + 1];
   282→    const h01 = this.hg[(iy + 1) * this.hgw + ix];
   283→    const h11 = this.hg[(iy + 1) * this.hgw + ix + 1];
   284→    return (h00 * (1 - fx) + h10 * fx) * (1 - fy) + (h01 * (1 - fx) + h11 * fx) * fy;
   285→  }

   289→  riverWidth = 30;

   291→  isWater(x: number, y: number): boolean {

   295→  /** shallow ford — vehicles may wade across, slowly */
   296→  fordAt(x: number, y: number, r = 46): boolean {
   297→    if (!this.ford) return false;
   298→    return dist(x, y, this.ford.x, this.ford.y) < r;
   299→  }

   301→  /** bridge within radius */
   302→  bridgeAt(x: number, y: number, r = 42): Bridge | null {
   303→    for (const b of this.bridges) {
   304→      if (dist(x, y, b.x, b.y) < r) return b;
   305→    }
   306→    return null;
   307→  }

   311→  roadFactor(x: number, y: number): number {
   312→    // 0 = no road, 1 = on major road centreline (approx)
   313→    let best = 0;
   314→    for (const r of this.roads) {
   315→      const d = this.distToPolyline(x, y, r.pts);
   316→      const w = r.major ? 26 : 20;
   317→      if (d < w) best = Math.max(best, (r.major ? 1 : 0.7) * (1 - d / w * 0.4));
   318→    }
   319→    return best;
   320→  }
   321→
   322→  railFactor(x: number, y: number): number {
   323→    const d = this.distToPolyline(x, y, this.railway);
   324→    return d < 14 ? 1 - d / 14 * 0.5 : 0;
   325→  }
   326→
   327→  forestDensity(x: number, y: number): number {
   328→    const n = this.forestNoise.fbm(x / 520, y / 520, 4);
   329→    // NW-biased woodland belt; the valley slopes carry brush
   330→    const west = 1 - x / this.W;
   331→    const bias = -0.075 * west + 0.035 * (y / this.H);
   332→    const v = (n - 0.585 + bias) * 4.2;
   333→    return clamp(v, 0, 1) * 1.6;
   334→  }

   901→  /** exact rotated-rect footprint test — used for LOS and cover */
   902→  buildingFootprintAt(x: number, y: number): Building | null {
   903→    const near = this.buildingsNear(x, y, 4);
   904→    for (const b of near) {
   905→      const dx = x - b.x;
   906→      const dy = y - b.y;
   907→      const c = Math.cos(-b.rot);
   908→      const s = Math.sin(-b.rot);
   909→      const lx = dx * c - dy * s;
   910→      const ly = dx * s + dy * c;
   911→      if (Math.abs(lx) <= b.w / 2 && Math.abs(ly) <= b.h / 2) return b;
   912→    }
   913→    return null;
   914→  }

   945→    let best: StoneWall | null = null;
   946→    let bd = r;
   947→    for (const w of this.walls) {
   948→      const d = dist(x, y, w.x, w.y);
   949→      if (d < Math.max(w.len * 0.5, bd) && d < bd) {
   950→        // rough test against the wall segment
   951→        const c = Math.cos(w.rot);
   952→        const s = Math.sin(w.rot);
   953→        const dx = x - w.x;
   954→        const dy = y - w.y;
   955→        const lx = dx * c + dy * s;
   956→        const ly = -dx * s + dy * c;
   957→        if (Math.abs(lx) <= w.len / 2 + 3 && Math.abs(ly) <= 10) {
   958→          bd = Math.max(2, Math.abs(ly));
   959→          best = w;
   960→        }
   961→      }
   962→    }
   963→    return best;
   964→  }

   968→  private extractContours(): ContourSet {
   969→    const step = 16; // metres between samples
   970→    const nx = Math.floor(this.W / step) + 1;
   971→    const ny = Math.floor(this.H / step) + 1;
   972→    const heights = new Float32Array(nx * ny);
   973→    for (let j = 0; j < ny; j++) {
   974→      for (let i = 0; i < nx; i++) {
   975→        heights[j * nx + i] = this.heightAt(i * step, j * step);
   976→      }
   977→    }
   978→    const minor = new Path2D();
   979→    const major = new Path2D();
   980→    const interval = 8;
   981→    const majorEvery = 5;
   982→
   983→    const lerpPt = (x1: number, y1: number, x2: number, y2: number, h1: number, h2: number, level: number) => {
   984→      const t = (level - h1) / (h2 - h1 || 1e-6);
   985→      return { x: (x1 + (x2 - x1) * t) * step, y: (y1 + (y2 - y1) * t) * step };
   986→    };
   987→
   988→    for (let j = 0; j < ny - 1; j++) {
   989→      for (let i = 0; i < nx - 1; i++) {
   990→        const x = i * step;
   991→        const y = j * step;
   992→        const h00 = heights[j * nx + i];
   993→        const h10 = heights[j * nx + i + 1];
   994→        const h01 = heights[(j + 1) * nx + i];
   995→        const h11 = heights[(j + 1) * nx + i + 1];
   996→        const hMin = Math.min(h00, h10, h01, h11);
   997→        const hMax = Math.max(h00, h10, h01, h11);
   998→        const firstLevel = Math.ceil(hMin / interval) * interval;
   999→        for (let level = firstLevel; level <= hMax; level += interval) {
  1000→          if (level <= 0) continue;
  1001→          const isMajor = Math.round(level / interval) % majorEvery === 0;
  1002→          const path = isMajor ? major : minor;
  1003→          // marching squares on cell corners
  1004→          let idx = 0;
  1005→          if (h00 >= level) idx |= 1;
  1006→          if (h10 >= level) idx |= 2;
  1007→          if (h11 >= level) idx |= 4;
  1008→          if (h01 >= level) idx |= 8;
  1009→          if (idx === 0 || idx === 15) continue;
  1010→          const top = () => lerpPt(i, j, i + 1, j, h00, h10, level);
  1011→          const right = () => lerpPt(i + 1, j, i + 1, j + 1, h10, h11, level);
  1012→          const bottom = () => lerpPt(i, j + 1, i + 1, j + 1, h01, h11, level);
  1013→          const left = () => lerpPt(i, j, i, j + 1, h00, h01, level);
  1014→          const seg = (a: { x: number; y: number }, b: { x: number; y: number }) => {
  1015→            path.moveTo(a.x, a.y);
  1016→            path.lineTo(b.x, b.y);
  1017→          };
  1018→          switch (idx) {
  1019→            case 1:
  1020→            case 14:
  1021→              seg(left(), top());
  1022→              break;
  1023→            case 2:
  1024→            case 13:
  1025→              seg(top(), right());
  1026→              break;
  1027→            case 3:
  1028→            case 12:
  1029→              seg(left(), right());
  1030→              break;
  1031→            case 4:
  1032→            case 11:
  1033→              seg(right(), bottom());
  1034→              break;
  1035→            case 6:
  1036→            case 9:
  1037→              seg(top(), bottom());
  1038→              break;
  1039→            case 7:
  1040→            case 8:
  1041→              seg(left(), bottom());
  1042→              break;
  1043→            case 5:
  1044→              seg(left(), top());
  1045→              seg(right(), bottom());
  1046→              break;
  1047→            case 10:
  1048→              seg(top(), right());
  1049→              seg(left(), bottom());
  1050→              break;
  1051→            default:
  1052→              break;
  1053→          }
  1054→        }
  1055→      }
  1056→    }
  1057→    return { minor, major };
  1058→  }

  1062→  private buildCostGrid() {
  1063→    this.gw = Math.ceil(this.W / CELL);
  1064→    this.gh = Math.ceil(this.H / CELL);
  1065→    const n = this.gw * this.gh;
  1066→    this.cost = new Float32Array(n);
  1067→    for (let gy = 0; gy < this.gh; gy++) {
  1068→      for (let gx = 0; gx < this.gw; gx++) {
  1069→        const cx = gx * CELL + CELL / 2;
  1070→        const cy = gy * CELL + CELL / 2;
  1071→        let c: number;
  1072→        if (this.isWater(cx, cy) && !this.bridgeAt(cx, cy, CELL * 0.9) && !this.fordAt(cx, cy, CELL)) {
  1073→          c = -1; // impassable water
  1074→        } else if (this.fordAt(cx, cy, CELL)) {
  1075→          c = 3.6; // wading — slow but possible
  1076→        } else {
  1077→          c = 1;
  1078→          c += this.slopeAt(cx, cy) * 14;
  1079→          const f = this.forestDensity(cx, cy);
  1080→          if (f > 0.42) c += 2.4;
  1081→          const r = this.roadFactor(cx, cy);
  1082→          if (r > 0.05) c = Math.min(c, r > 0.5 ? 0.34 : 0.6);
  1083→          if (this.railFactor(cx, cy) > 0.4) c += 1.2;
  1084→          if (this.buildingAt(cx, cy, 10)) c += 5;
  1085→          if (this.trenchDist(cx, cy) < 8) c += 1.4; // crossing a dug position
  1086→          if (this.bridgeAt(cx, cy, CELL * 0.9)) c = 0.5;
  1087→        }
  1088→        this.cost[gy * this.gw + gx] = c;
  1089→      }
  1090→    }
  1091→  }

  1093→  private buildHeightGrid() {
  1094→    this.hgw = Math.ceil(this.W / HG_STEP) + 1;
  1095→    this.hgh = Math.ceil(this.H / HG_STEP) + 1;
  1096→    this.hg = new Float32Array(this.hgw * this.hgh);
  1097→    for (let j = 0; j < this.hgh; j++) {
  1098→      for (let i = 0; i < this.hgw; i++) {
  1099→        this.hg[j * this.hgw + i] = this.heightAt(i * HG_STEP, j * HG_STEP);
  1100→      }
  1101→    }
  1102→    // building occupancy mask — rasterize every footprint once
  1103→    this.bmw = Math.ceil(this.W / this.BM_CELL) + 1;
  1104→    this.bmh = Math.ceil(this.H / this.BM_CELL) + 1;
  1105→    this.bmask = new Uint8Array(this.bmw * this.bmh);
  1106→    for (const b of this.buildings) {
  1107→      if (b.kind === 'MAST' || b.kind === 'CHECKPOINT') continue; // see-through
  1108→      const c = Math.cos(b.rot);
  1109→      const s = Math.sin(b.rot);
  1110→      const reach = Math.max(b.w, b.h) * 0.5;
  1111→      const gx0 = Math.max(0, ((b.x - reach) / this.BM_CELL) | 0);
  1112→      const gx1 = Math.min(this.bmw - 1, ((b.x + reach) / this.BM_CELL) | 0);
  1113→      const gy0 = Math.max(0, ((b.y - reach) / this.BM_CELL) | 0);
  1114→      const gy1 = Math.min(this.bmh - 1, ((b.y + reach) / this.BM_CELL) | 0);
  1115→      for (let gy = gy0; gy <= gy1; gy++) {
  1116→        for (let gx = gx0; gx <= gx1; gx++) {
  1117→          const px = gx * this.BM_CELL + this.BM_CELL / 2;
  1118→          const py = gy * this.BM_CELL + this.BM_CELL / 2;
  1119→          const dx = px - b.x;
  1120→          const dy = py - b.y;
  1121→          const lx = dx * c + dy * s;
  1122→          const ly = -dx * s + dy * c;
  1123→          if (Math.abs(lx) <= b.w / 2 && Math.abs(ly) <= b.h / 2) {
  1124→            this.bmask[gy * this.bmw + gx] = 1;
  1125→          }
  1126→        }
  1127→      }
  1128→    }
  1129→  }

  1131→  findPath(from: Vec2, to: Vec2): Vec2[] {
  1132→    const sgx = clamp(Math.floor(from.x / CELL), 0, this.gw - 1);
  1133→    const sgy = clamp(Math.floor(from.y / CELL), 0, this.gh - 1);
  1134→    let tgx = clamp(Math.floor(to.x / CELL), 0, this.gw - 1);
  1135→    let tgy = clamp(Math.floor(to.y / CELL), 0, this.gh - 1);
  1136→    if (this.cost[tgy * this.gw + tgx] < 0) {
  1137→      // nudge target to nearest passable cell
  1138→      const alt = this.nearestPassable(tgx, tgy);
  1139→      if (!alt) return [{ ...to }];
  1140→      tgx = alt[0];
  1141→      tgy = alt[1];
  1142→    }
  1143→    if (sgx === tgx && sgy === tgy) return [{ ...to }];
  1144→
  1145→    const n = this.gw * this.gh;
  1146→    const gScore = new Float32Array(n).fill(Infinity);
  1147→    const cameFrom = new Int32Array(n).fill(-1);
  1148→    const closed = new Uint8Array(n);
  1149→    const open: number[] = [];
  1150→    const fScore = new Float32Array(n).fill(Infinity);
  1151→    const h = (x: number, y: number) => (Math.abs(x - tgx) + Math.abs(y - tgy)) * 0.6;
  1152→    const start = sgy * this.gw + sgx;
  1153→    gScore[start] = 0;
  1154→    fScore[start] = h(sgx, sgy);
  1155→    open.push(start);
  1156→    let guard = 0;
  1157→    while (open.length && guard++ < 12000) {
  1158→      // find best in open
  1159→      let bi = 0;
  1160→      for (let i = 1; i < open.length; i++) {
  1161→        if (fScore[open[i]] < fScore[open[bi]]) bi = i;
  1162→      }
  1163→      const cur = open.splice(bi, 1)[0];
  1164→      if (cur === tgy * this.gw + tgx) break;
  1165→      closed[cur] = 1;
  1166→      const cx = cur % this.gw;
  1167→      const cy = (cur / this.gw) | 0;
  1168→      for (let dy = -1; dy <= 1; dy++) {
  1169→        for (let dx = -1; dx <= 1; dx++) {
  1170→          if (!dx && !dy) continue;
  1171→          const nx = cx + dx;
  1172→          const ny = cy + dy;
  1173→          if (nx < 0 || ny < 0 || nx >= this.gw || ny >= this.gh) continue;
  1174→          const ni = ny * this.gw + nx;
  1175→          if (closed[ni] || this.cost[ni] < 0) continue;
  1176→          const step = (dx && dy ? 1.414 : 1) * (this.cost[ni] + this.cost[cur]) * 0.5;
  1177→          const tentative = gScore[cur] + step;
  1178→          if (tentative < gScore[ni]) {
  1179→            gScore[ni] = tentative;
  1180→            cameFrom[ni] = cur;
  1181→            fScore[ni] = tentative + h(nx, ny);
  1182→            if (!open.includes(ni)) open.push(ni);
  1183→          }
  1184→        }
  1185→      }
  1186→    }
  1187→
  1188→    const end = tgy * this.gw + tgx;
  1189→    if (cameFrom[end] === -1 && end !== start) return [{ ...to }];
  1190→    // reconstruct
  1191→    const cells: number[] = [];
  1192→    let c = end;
  1193→    while (c !== -1 && c !== start) {
  1194→      cells.push(c);
  1195→      c = cameFrom[c];
  1196→    }
  1197→    cells.push(start);
  1198→    cells.reverse();
  1199→    // waypoints at cell centres, then string-pull
  1200→    let pts: Vec2[] = cells.map((ci) => ({
  1201→      x: (ci % this.gw) * CELL + CELL / 2,
  1202→      y: ((ci / this.gw) | 0) * CELL + CELL / 2,
  1203→    }));
  1204→    pts.push({ ...to });
  1205→    pts = this.smoothPath(pts);
  1206→    return pts.length ? pts : [{ ...to }];
  1207→  }

  1209→  private nearestPassable(gx: number, gy: number): [number, number] | null {
  1210→    for (let r = 1; r < 8; r++) {
  1211→      for (let dy = -r; dy <= r; dy++) {
  1212→        for (let dx = -r; dx <= r; dx++) {
  1213→          const nx = gx + dx;
  1214→          const ny = gy + dy;
  1215→          if (nx < 0 || ny < 0 || nx >= this.gw || ny >= this.gh) continue;
  1216→          if (this.cost[ny * this.gw + nx] >= 0) return [nx, ny];
  1217→        }
  1218→      }
  1219→    }
  1220→    return null;
  1221→  }

  1223→  private smoothPath(pts: Vec2[]): Vec2[] {
  1224→    if (pts.length < 3) return pts;
  1225→    const out: Vec2[] = [pts[0]];
  1226→    let i = 0;
  1227→    while (i < pts.length - 1) {
  1228→      let j = pts.length - 1;
  1229→      for (; j > i + 1; j--) {
  1230→        if (this.los(pts[i], pts[j])) break;
  1231→      }
  1232→      out.push(pts[j]);
  1233→      i = j;
  1234→    }
  1235→    return out;
  1236→  }

  1238→  los(a: Vec2, b: Vec2): boolean {
  1239→    const d = dist(a.x, a.y, b.x, b.y);
  1240→    const steps = Math.max(2, Math.ceil(d / (CELL * 0.6)));
  1241→    for (let s = 1; s < steps; s++) {
  1242→      const t = s / steps;
  1243→      const x = a.x + (b.x - a.x) * t;
  1244→      const y = a.y + (b.y - a.y) * t;
  1245→      if (this.isWater(x, y) && !this.bridgeAt(x, y, 30) && !this.fordAt(x, y, 40)) return false;
  1246→      if (this.slopeAt(x, y) > 0.75) return false;
  1247→    }
  1248→    return true;
  1249→  }

  1272→  distToPolyline(x: number, y: number, pts: Vec2[]): number {
  1273→    let best = Infinity;
  1274→    for (let i = 0; i < pts.length - 1; i++) {
  1275→      const a = pts[i];
  1276→      const b = pts[i + 1];
  1277→      const vx = b.x - a.x;
  1278→      const vy = b.y - a.y;
  1279→      const wx = x - a.x;
  1280→      const wy = y - a.y;
  1281→      const len2 = vx * vx + vy * vy || 1e-6;
  1282→      let t = (wx * vx + wy * vy) / len2;
  1283→      t = clamp(t, 0, 1);
  1284→      const dx = x - (a.x + vx * t);
  1285→      const dy = y - (a.y + vy * t);
  1286→      const d = dx * dx + dy * dy;
  1287→      if (d < best) best = d;
  1288→    }
  1289→    return Math.sqrt(best);
  1290→  }

  1292→  polylineDirAt(x: number, y: number, pts: Vec2[]): number {
  1293→    let best = Infinity;
  1294→    let bestI = 0;
  1295→    let bestT = 0;
  1296→    for (let i = 0; i < pts.length - 1; i++) {
  1297→      const a = pts[i];
  1298→      const b = pts[i + 1];
  1299→      const vx = b.x - a.x;
  1300→      const vy = b.y - a.y;
  1301→      const wx = x - a.x;
  1302→      const wy = y - a.y;
  1303→      const len2 = vx * vx + vy * vy || 1e-6;
  1304→      let t = (wx * vx + wy * vy) / len2;
  1305→      t = clamp(t, 0, 1);
  1306→      const dx = x - (a.x + vx * t);
  1307→      const dy = y - (a.y + vy * t);
  1308→      const d = dx * dx + dy * dy;
  1309→      if (d < best) {
  1310→        best = d;
  1311→        bestI = i;
  1312→        bestT = t;
  1313→      }
  1314→    }
  1315→    const a = pts[bestI];
  1316→    const b = pts[bestI + 1];
  1317→    void bestT;
  1318→    return Math.atan2(b.y - a.y, b.x - a.x);
  1319→  }
