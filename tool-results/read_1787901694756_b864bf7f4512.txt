   194→  private approachTarget(target: Unit, ctx: SimContext) {
   195→    const d = dist(this.x, this.y, target.x, target.y);
   196→    if (d > this.def.range * 0.85) {
   197→      const t = 0.8;
   198→      const px = this.x + (target.x - this.x) * t;
   199→      const py = this.y + (target.y - this.y) * t;
   200→      this.path = ctx.terrain.findPath({ x: this.x, y: this.y }, { x: px, y: py });
   201→      this.dest = { x: px, y: py };
   202→      this.pathIndexStart();
   203→    }
   204→  }
   205→
   206→  private pathIndexStart() {
   207→    // drop the first waypoint if it is behind us
   208→    if (this.path.length > 1) {
   209→      const a = this.path[0];
   210→      const b = this.path[1];
   211→      const toB = angleOf(b.x - this.x, b.y - this.y);
   212→      const toA = angleOf(a.x - this.x, a.y - this.y);
   213→      if (Math.abs(angDiff(toA, toB)) > Math.PI * 0.6) this.path.shift();
   214→    }
   215→  }

   217→  // ── main update ────────────────────────────────────────────
   218→
   219→  update(dt: number, ctx: SimContext) {
   220→    if (this.dead) return;
   221→    if (this.def.kind === 'FACTORY') {
   222→      // static structure — no movement, no weapons
   223→      this.damageFlash = Math.max(0, this.damageFlash - dt * 3);
   224→      return;
   225→    }
   226→    this.reloadT -= dt;
   227→    this.damageFlash = Math.max(0, this.damageFlash - dt * 3);
   228→    this.recoil = Math.max(0, this.recoil - dt * 8);
   229→    if (this.def.kind === 'SPAA') this.radarAngle += dt * 1.35;
   230→
   231→    // suppression decays — troops recover their nerve
   232→    this.suppression = Math.max(0, this.suppression - dt * 0.065);
   233→    this.supSeekCooldown -= dt;
   234→    this.standoffCheckT -= dt;
   235→    this.reconCheckT -= dt;
   236→
   237→    if (this.isAir) {
   238→      this.updateAir(dt, ctx);
   239→      return;
   240→    }
   241→
   242→    this.updateRoleBehavior(dt, ctx);
   243→    this.updateGroundMovement(dt, ctx);
   244→    this.updateTargeting(dt, ctx);
   245→    this.updateFiring(dt, ctx);
   246→    this.updateSurfaceFx(dt, ctx);
   247→  }
   248→
   249→  get pinned(): boolean {
   250→    return this.suppression > 0.85;
   251→  }

   253→  /** aim quality multiplier — everything tactical converges here */
   254→  accuracyVs(t: Unit, ctx: SimContext): number {
   255→    let m = 1;
   256→    // firing on the move degrades gunnery
   257→    if (this.speedNow > this.def.speed * 0.25) m *= 0.72;
   258→    // a suppressed crew flinches
   259→    m *= 1 - 0.45 * this.suppression;
   260→    // high ground observes better
   261→    const dh = ctx.terrain.heightAt(this.x, this.y) - ctx.terrain.heightAt(t.x, t.y);
   262→    if (dh > 10) m *= 1.15;
   263→    else if (dh < -10) m *= 0.9;
   264→    // target movement
   265→    if (t.speedNow > t.def.speed * 0.5) m *= 0.82;
   266→    else if (t.speedNow < 0.5) m *= 1.08;
   267→    // cover on the target — the decisive factor
   268→    const cov = coverFrom(ctx, t.x, t.y, this.x, this.y);
   269→    m *= 1 - cov.value;
   270→    return clamp(m, 0.12, 1.25);
   271→  }

   273→  /** roles behave like they want to survive */
   274→  private updateRoleBehavior(dt: number, ctx: SimContext) {
   275→    void dt;
   276→    // artillery keeps distance from the fight
   277→    if (this.def.kind === 'SPG' && this.standoffCheckT <= 0) {
   278→      this.standoffCheckT = 4 + this.rng.range(0, 2);
   279→      let nearest: Unit | null = null;
   280→      let nd = Infinity;
   281→      for (const u of ctx.units) {
   282→        if (u.dead || u.faction === this.faction || u.isAir) continue;
   283→        const d = dist(this.x, this.y, u.x, u.y);
   284→        if (d < nd) {
   285→          nd = d;
   286→          nearest = u;
   287→        }
   288→      }
   289→      if (nearest && nd < 750 && this.fireMissionLeft <= 0) {
   290→        const away = angleOf(this.x - nearest.x, this.y - nearest.y);
   291→        const dest = {
   292→          x: clamp(this.x + Math.cos(away) * 620, 100, ctx.terrain.W - 100),
   293→          y: clamp(this.y + Math.sin(away) * 620, 100, ctx.terrain.H - 100),
   294→        };
   295→        this.orderMove(dest, ctx);
   296→        ctx.log(`${this.callsign} · DISPLACING — ENEMY INSIDE GUN RANGE`, 'alert');
   297→      }
   298→    }
   299→
   300→    // reconnaissance survives to scout again
   301→    if (this.def.kind === 'REC' && this.reconCheckT <= 0) {
   302→      this.reconCheckT = 3 + this.rng.range(0, 2);
   303→      let threat = 0;
   304→      let nearest: Unit | null = null;
   305→      let nd = Infinity;
   306→      for (const u of ctx.units) {
   307→        if (u.dead || u.faction === this.faction || u.isAir) continue;
   308→        if (u.def.kind === 'FACTORY' || u.def.kind === 'HQ') continue;
   309→        const d = dist(this.x, this.y, u.x, u.y);
   310→        if (d < 900) threat++;
   311→        if (d < nd) {
   312→          nd = d;
   313→          nearest = u;
   314→        }
   315→      }
   316→      const outnumbered = threat >= 2 && nearest && nd < 520;
   317→      // a player-given move order is respected — the scout flinches only
   318→      // when idle, or when it is genuinely fighting for its life
   319→      const executingOrder = (this.order.type === 'MOVE' || this.order.type === 'ATTACK_MOVE') && this.path.length > 0;
   320→      const mustBreak = this.hp < this.def.hp * 0.55 || (outnumbered && !executingOrder);
   321→      if (mustBreak && threat > 0) {
   322→        // fall back toward the nearest friendly armoured unit
   323→        let haven: Unit | null = null;
   324→        let hd = Infinity;
   325→        for (const u of ctx.units) {
   326→          if (u.dead || u.faction !== this.faction || u.isAir) continue;
   327→          if (u.def.kind !== 'MBT' && u.def.kind !== 'IFV') continue;
   328→          const d = dist(this.x, this.y, u.x, u.y);
   329→          if (d < hd) {
   330→            hd = d;
   331→            haven = u;
   332→          }
   333→        }
   334→        if (haven && hd > 140) {
   335→          const away = angleOf(haven.x - this.x, haven.y - this.y);
   336→          this.orderMove(
   337→            { x: haven.x + Math.cos(away) * 40, y: haven.y + Math.sin(away) * 40 },
   338→            ctx
   339→          );
   340→          ctx.log(`${this.callsign} · WITHDRAWING UNDER PRESSURE`, 'alert');
   341→        }
   342→      }
   343→    }
   344→
   345→    // under fire with no orders: seek nearby cover (both factions)
   346→    if (
   347→      this.suppression > 0.35 &&
   348→      this.supSeekCooldown <= 0 &&
   349→      (this.order.type === 'HOLD' || this.order.type === 'STOP') &&
   350→      this.path.length === 0 &&
   351→      this.lastAttacker &&
   352→      !this.lastAttacker.dead &&
   353→      ctx.time - this.lastAttackedT < 14
   354→    ) {
   355→      this.supSeekCooldown = 7;
   356→      const spot = findCoverSpot(ctx, this.x, this.y, this.lastAttacker.x, this.lastAttacker.y, 90);
   357→      if (spot && spot.value > 0.3) {
   358→        const current = coverFrom(ctx, this.x, this.y, this.lastAttacker.x, this.lastAttacker.y);
   359→        if (spot.value > current.value + 0.15) {
   360→          this.path = [{ x: spot.x, y: spot.y }];
   361→          this.dest = { x: spot.x, y: spot.y };
   362→        }
   363→      }
   364→    }
   365→  }

   369→  private updateGroundMovement(dt: number, ctx: SimContext) {
   370→    // ATTACK order: keep approaching if target far
   371→    if (this.order.type === 'ATTACK' && this.target && !this.target.dead) {
   372→      const d = dist(this.x, this.y, this.target.x, this.target.y);
   373→      if (d > this.def.range * 0.9 && (this.path.length === 0 || !this.dest)) {
   374→        this.approachTarget(this.target, ctx);
   375→      }
   376→      if (d < this.def.range * 0.55 && this.stance === 'HOLD') {
   377→        this.path = [];
   378→        this.dest = null;
   379→      }
   380→    }
   381→    // FIRE_MISSION: hold position
   382→    if (this.order.type === 'FIRE_MISSION') {
   383→      this.path = [];
   384→      this.dest = null;
   385→    }
   386→
   387→    if (this.path.length === 0) {
   388→      // decelerate
   389→      this.speedNow = Math.max(0, this.speedNow - dt * this.def.speed * 1.6);
   390→      // rotate turret home slowly
   391→      if (!this.target) {
   392→        this.turretAngle = rotateToward(this.turretAngle, this.angle, this.def.turretRate * 0.35 * dt);
   393→      }
   394→      return;
   395→    }
   396→
   397→    const wp = this.path[0];
   398→    const d = dist(this.x, this.y, wp.x, wp.y);
   399→    const arriveR = this.path.length === 1 ? 5 : 14;
   400→    if (d < arriveR) {
   401→      this.path.shift();
   402→      if (this.path.length === 0) {
   403→        this.dest = null;
   404→        this.speedNow = 0;
   405→        this.isReinforcement = false;
   406→      }
   407→      return;
   408→    }
   409→
   410→    const desired = angleOf(wp.x - this.x, wp.y - this.y);
   411→    const diff = angDiff(this.angle, desired);
   412→    const turnCap = this.def.turnRate * dt;
   413→    this.angle = rotateToward(this.angle, desired, turnCap);
   414→
   415→    // slow down on sharp turns
   416→    const turnFactor = 1 - clamp(Math.abs(diff) / Math.PI, 0, 1) * 0.75;
   417→    // terrain speed
   418→    let terr = 1;
   419→    const forest = ctx.terrain.forestDensity(this.x, this.y);
   420→    if (forest > 0.42) terr *= 0.58;
   421→    const slope = ctx.terrain.slopeAt(this.x, this.y);
   422→    terr *= 1 / (1 + slope * 2.6);
   423→    const road = ctx.terrain.roadFactor(this.x, this.y);
   424→    if (road > 0.3) terr *= 1.25;
   425→
   426→    const targetSpeed =
   427→      this.def.speed * turnFactor * clamp(terr, 0.3, 1.35) * (1 - 0.35 * this.suppression) * (this.pinned ? 0.5 : 1);
   428→    this.speedNow += clamp(targetSpeed - this.speedNow, -this.def.speed * 2 * dt, this.def.speed * 1.4 * dt);
   429→
   430→    const step = this.speedNow * dt;
   431→    this.x += Math.cos(this.angle) * step;
   432→    this.y += Math.sin(this.angle) * step;
   433→
   434→    // separation from nearby units — vehicles keep fighting distance
   435→    for (const o of ctx.units) {
   436→      if (o === this || o.dead || o.isAir) continue;
   437→      const od = dist(this.x, this.y, o.x, o.y);
   438→      const heavy = this.def.kind === 'MBT' || o.def.kind === 'MBT';
   439→      const minD = (this.def.length + o.def.length) * 0.42 + (heavy ? 20 : 12);
   440→      if (od < minD && od > 0.01) {
   441→        const push = ((minD - od) / minD) * 30 * dt;
   442→        const a = angleOf(this.x - o.x, this.y - o.y);
   443→        this.x += Math.cos(a) * push;
   444→        this.y += Math.sin(a) * push;
   445→      }
   446→    }
   447→
   448→    // stuck detection: nudge sideways
   449→    if (this.speedNow < 0.6 && this.path.length > 0) {
   450→      this.stuckT += dt;
   451→      if (this.stuckT > 2.4) {
   452→        this.stuckT = 0;
   453→        const side = this.rng.chance(0.5) ? 1 : -1;
   454→        this.angle += side * 0.7;
   455→        this.x += Math.cos(this.angle) * 3;
   456→        this.y += Math.sin(this.angle) * 3;
   457→      }
   458→    } else {
   459→      this.stuckT = 0;
   460→    }
   461→
   462→    this.x = clamp(this.x, 20, ctx.terrain.W - 20);
   463→    this.y = clamp(this.y, 20, ctx.terrain.H - 20);
   464→  }

   466→  // ── targeting ──────────────────────────────────────────────
   467→
   468→  private updateTargeting(dt: number, ctx: SimContext) {
   469→    void dt;
   470→    // validate current target
   471→    if (this.target && (this.target.dead || !this.sees(this.target))) {
   472→      this.target = null;
   473→    }
   474→    // pick new target from currently visible enemies
   475→    if (!this.target || this.target.dead) {
   476→      let best: Unit | null = null;
   477→      let bestScore = -Infinity;
   478→      for (const u of this.visibleTargets) {
   479→        if (u.dead || u.faction === this.faction) continue;
   480→        // factories are engaged on explicit order only — no one wastes
   481→        // main gun rounds on empty halls by accident
   482→        if (u.def.kind === 'FACTORY') continue;
   483→        let s = 100 - dist(this.x, this.y, u.x, u.y) / 30;
   484→        // role preferences
   485→        if (this.def.kind === 'MBT') s += u.def.kind === 'MBT' ? 24 : u.def.kind === 'SPAA' ? 30 : 0;
   486→        if (this.def.kind === 'IFV' || this.def.kind === 'REC') s += u.def.kind === 'REC' ? 18 : u.def.kind === 'IFV' ? 10 : 0;
   487→        if (this.def.kind === 'SPAA') continue; // air defence does not engage ground
   488→        if (u.isAir && !this.def.canHitAir) continue;
   489→        if (s > bestScore) {
   490→          bestScore = s;
   491→          best = u;
   492→        }
   493→      }
   494→      if (best) this.target = best;
   495→    }
   496→    // turret
   497→    if (this.target) {
   498→      const lead = this.def.projectile === 'SHELL' ? this.leadAngle(this.target, 470) : angleOf(this.target.x - this.x, this.target.y - this.y);
   499→      this.turretAngle = rotateToward(this.turretAngle, lead, this.def.turretRate * (1 - 0.5 * this.suppression) * dt);
   500→    }
   501→  }
   502→
   503→  private leadAngle(target: Unit, projSpeed: number): number {
   504→    const d = dist(this.x, this.y, target.x, target.y);
   505→    const t = d / projSpeed;
   506→    const px = target.x + Math.cos(target.angle) * target.speedNow * t;
   507→    const py = target.y + Math.sin(target.angle) * target.speedNow * t;
   508→    return angleOf(px - this.x, py - this.y);
   509→  }
   510→
   511→  sees(u: Unit): boolean {
   512→    return this.visibleTargets.includes(u);
   513→  }

   515→  // ── firing ─────────────────────────────────────────────────
   516→
   517→  private updateFiring(dt: number, ctx: SimContext) {
   518→    // artillery fire missions (player-directed or AI-directed)
   519→    if (this.def.projectile === 'ARTY') {
   520→      if (this.fireMissionArea && this.fireMissionLeft > 0 && this.reloadT <= 0 && this.ammo > 0 && this.speedNow < 0.6) {
   521→        let ax = this.fireMissionArea.x;
   522→        let ay = this.fireMissionArea.y;
   523→        let quality = 1; // dispersion multiplier — lower is better
   524→        // corrected fire on a tracked target: live observation tightens the salvo
   525→        if (this.artyTrack && !this.artyTrack.dead) {
   526→          const observed = this.artyTrack.faction === 'ENEMY'
   527→            ? this.artyTrack.intel === 'DETECTED'
   528→            : ctx.units.some(
   529→                (u) => u.faction === 'ENEMY' && !u.dead && u.visibleTargets.includes(this.artyTrack!)
   530→              );
   531→          if (observed) {
   532→            ax = this.artyTrack.x;
   533→            ay = this.artyTrack.y;
   534→            quality = 0.85;
   535→            if (this.artyTrack.speedNow > 2) quality *= 1.6; // a moving target slips the bracket
   536→          } else {
   537→            ax = this.artyTrack.knownX;
   538→            ay = this.artyTrack.knownY;
   539→            quality = 2.4;
   540→            if (ctx.time - this.artyTrack.lastSeen > 30) this.artyTrack = null;
   541→          }
   542→        } else {
   543→          // area fire — is anyone observing the target area?
   544→          const observed = ctx.units.some(
   545→            (u) =>
   546→              !u.dead &&
   547→              u.faction === this.faction &&
   548→              !u.isAir &&
   549→              dist(u.x, u.y, ax, ay) < Math.max(600, u.def.vision * 0.65) &&
   550→              ctx.terrain.losClear(u.x, u.y, ctx.terrain.heightAt(u.x, u.y) + 4.5, ax, ay, ctx.terrain.heightAt(ax, ay) + 3)
   551→          );
   552→          quality = observed ? 1.35 : 4.0;
   553→        }
   554→        // shooter state widens the bracket
   555→        if (this.suppression > 0.3) quality *= 1 + (this.suppression - 0.3);
   556→        if (this.hp < this.def.hp * 0.5) quality *= 1.3;
   557→        // distance tells
   558→        const dFire = dist(this.x, this.y, ax, ay);
   559→        quality *= 1 + (dFire - 800) / 8000;
   560→
   561→        ctx.projectiles.fireArtillery(ctx, this, ax, ay, this.def.damage, this.def.splash, quality);
   562→        this.ammo--;
   563→        this.fireMissionLeft--;
   564→        this.reloadT = this.def.reload;
   565→        this.recoil = 1.4;
   566→        this.lastFireT = ctx.time;
   567→        if (this.fireMissionLeft === 0) {
   568→          ctx.log(`${this.callsign} · FIRE MISSION COMPLETE`);
   569→          this.fireMissionArea = null;
   570→          this.artyTrack = null;
   571→        }
   572→        if (this.ammo === 0) ctx.log(`${this.callsign} · AMMUNITION EXPENDED`, 'alert');
   573→      }
   574→      return;
   575→    }
   576→
   577→    // burst weapons
   578→    if (this.burstLeft > 0) {
   579→      this.burstT -= dt;
   580→      if (this.burstT <= 0 && this.target && !this.target.dead) {
   581→        this.burstT = this.def.burstInterval;
   582→        this.burstLeft--;
   583→        ctx.projectiles.fireAuto(ctx, this, this.target.x, this.target.y, this.def.damage, this.accuracyVs(this.target, ctx));
   584→        this.ammo--;
   585→        this.recoil = 0.4;
   586→        this.lastFireT = ctx.time;
   587→        if (this.ammo <= 0) {
   588→          this.burstLeft = 0;
   589→          ctx.log(`${this.callsign} · AMMUNITION EXPENDED`, 'alert');
   590→        }
   591→      }
   592→      return;
   593→    }
   594→
   595→    if (!this.target || this.target.dead || this.reloadT > 0 || this.ammo <= 0) return;
   596→    // a pinned crew keeps its head down
   597→    if (this.pinned && this.rng.chance(0.6)) return;
   598→
   599→    const d = dist(this.x, this.y, this.target.x, this.target.y);
   600→    if (d > this.def.range || d < this.def.minRange) return;
   601→    if (this.target.isAir && !this.def.canHitAir) return;
   602→
   603→    const aimAngle = this.def.projectile === 'SHELL' ? this.leadAngle(this.target, 470) : angleOf(this.target.x - this.x, this.target.y - this.y);
   604→    const aligned = Math.abs(angDiff(this.turretAngle, aimAngle));
   605→
   606→    if (this.def.projectile === 'AUTO') {
   607→      if (aligned < 0.14) {
   608→        this.burstLeft = this.def.burst;
   609→        this.burstT = 0;
   610→        this.reloadT = this.def.reload;
   611→      }
   612→    } else if (this.def.projectile === 'SHELL') {
   613→      if (aligned < 0.05) {
   614→        const acc = this.accuracyVs(this.target, ctx);
   615→        ctx.projectiles.fireShell(ctx, this, this.target.x, this.target.y, this.def.damage, this.def.accuracy * acc);
   616→        this.ammo--;
   617→        this.reloadT = this.def.reload;
   618→        this.recoil = 1.0;
   619→        this.lastFireT = ctx.time;
   620→        if (this.ammo <= 0) ctx.log(`${this.callsign} · AMMUNITION EXPENDED`, 'alert');
   621→      }
   622→    } else if (this.def.projectile === 'MISSILE_SPAA') {
   623→      if (aligned < 0.2 && this.target.isAir) {
   624→        ctx.projectiles.fireSAM(ctx, this, this.target, this.def.damage);
   625→        this.ammo--;
   626→        this.reloadT = this.def.reload;
   627→        this.recoil = 0.6;
   628→        if (this.ammo <= 0) ctx.log(`ENEMY AD · SALVO EXHAUSTED`, 'info');
   629→      }
   630→    }
   631→  }

   656→  // ── aircraft ───────────────────────────────────────────────
   657→
   658→  private updateAir(dt: number, ctx: SimContext) {
   659→    const speed = this.def.speed;
   660→
   661→    switch (this.airState) {
   662→      case 'STANDBY':
   663→        this.rearmT -= dt;
   664→        break;
   665→
   666→      case 'INBOUND':
   667→      case 'PATROL': {
   668→        const speed = this.def.speed;
   669→
   670→        // ── committed attack runs: dive, fire, egress, re-attack ──
   671→        if (this.runPhase === 'EGRESS') {
   672→          this.egressT -= dt;
   673→          // fly through — the aircraft does not linger over the target
   674→          const away = this.runTarget && !this.runTarget.dead ? angleOf(this.x - this.runTarget.x, this.y - this.runTarget.y) : this.angle;
   675→          const desired = this.angle + angDiff(this.angle, away) * 0.22;
   676→          const turn = clamp(angDiff(this.angle, desired), -this.def.turnRate * dt, this.def.turnRate * dt);
   677→          this.angle += turn;
   678→          this.bank = clamp(this.bank + (turn / (this.def.turnRate * dt || 1) - this.bank) * dt * 2.5, -1, 1);
   679→          this.x += Math.cos(this.angle) * speed * dt;
   680→          this.y += Math.sin(this.angle) * speed * dt;
   681→          if (this.egressT <= 0) {
   682→            this.runPhase = 'ORBIT';
   683→            this.runTarget = null;
   684→          }
   685→          break;
   686→        }
   687→
   688→        if (this.runPhase === 'INGRESS' && this.runTarget && !this.runTarget.dead) {
   689→          // straight at the target — commitment
   690→          const toT = angleOf(this.runTarget.x - this.x, this.runTarget.y - this.y);
   691→          const diff = angDiff(this.angle, toT);
   692→          const turn = clamp(diff, -this.def.turnRate * dt, this.def.turnRate * dt);
   693→          this.angle += turn;
   694→          this.bank = clamp(this.bank + (turn / (this.def.turnRate * dt || 1) - this.bank) * dt * 3, -1, 1);
   695→          this.x += Math.cos(this.angle) * speed * dt;
   696→          this.y += Math.sin(this.angle) * speed * dt;
   697→          const dd = dist(this.x, this.y, this.runTarget.x, this.runTarget.y);
   698→          if (dd < this.def.range && Math.abs(diff) < 0.35 && this.ammo > 0 && this.reloadT <= 0) {
   699→            // weapons away — then get out
   700→            ctx.projectiles.fireAGM(ctx, this, this.runTarget, this.def.damage, this.def.splash);
   701→            this.ammo--;
   702→            this.reloadT = this.def.reload;
   703→            this.runPhase = 'EGRESS';
   704→            this.egressT = 3.4 + this.rng.range(0, 1.6);
   705→            if (this.ammo === 0) {
   706→              ctx.log(`${this.callsign} · ORDnANCE EXPENDED — RTB`, 'info');
   707→              this.airState = 'RTB';
   708→              this.runPhase = 'ORBIT';
   709→              this.runTarget = null;
   710→            }
   711→          } else if (dd > 2400 || (dd < 120 && Math.abs(diff) > 2.2)) {
   712→            // overshoot — abort and reset
   713→            this.runPhase = 'EGRESS';
   714→            this.egressT = 2.6;
   715→          }
   716→          break;
   717→        }
   718→
   719→        // orbit the patrol point
   720→        const R = 640;
   721→        const dx = this.x - this.patrol.x;
   722→        const dy = this.y - this.patrol.y;
   723→        const d = Math.hypot(dx, dy) || 1;
   724→        const orbitA = angleOf(dx, dy);
   725→        let desired: number;
   726→        if (d > R * 1.6) {
   727→          desired = angleOf(this.patrol.x - this.x, this.patrol.y - this.y);
   728→        } else {
   729→          // tangential heading, correct radius drift
   730→          const radialErr = (d - R) / R;
   731→          desired = orbitA + Math.PI / 2 * this.orbitDir + clamp(radialErr, -0.7, 0.7) * this.orbitDir;
   732→        }
   733→        const diff = angDiff(this.angle, desired);
   734→        const turn = clamp(diff, -this.def.turnRate * dt, this.def.turnRate * dt);
   735→        this.angle += turn;
   736→        this.bank = clamp(this.bank + (turn / (this.def.turnRate * dt || 1) - this.bank) * dt * 3, -1, 1);
   737→        this.x += Math.cos(this.angle) * speed * dt;
   738→        this.y += Math.sin(this.angle) * speed * dt;
   739→
   740→        // arrive on station
   741→        if (this.airState === 'INBOUND' && Math.hypot(this.x - this.patrol.x, this.y - this.patrol.y) < R * 1.6) {
   742→          this.airState = 'PATROL';
   743→        }
   744→
   745→        // pick a target and begin the dive
   746→        if (this.ammo > 0 && this.reloadT <= 0 && (this.airState === 'PATROL' || this.airState === 'INBOUND')) {
   747→          let best: Unit | null = null;
   748→          let bd = Infinity;
   749→          for (const u of this.visibleTargets) {
   750→            if (u.dead || u.isAir || u.def.kind === 'FACTORY') continue;
   751→            const dd = dist(this.x, this.y, u.x, u.y);
   752→            if (dd < 1900 && dd < bd) {
   753→              const toT = angleOf(u.x - this.x, u.y - this.y);
   754→              // prefer targets roughly ahead
   755→              const off = Math.abs(angDiff(this.angle, toT));
   756→              const score = dd + off * 400;
   757→              if (score < bd) {
   758→                bd = score;
   759→                best = u;
   760→              }
   761→            }
   762→          }
   763→          if (best) {
   764→            this.runTarget = best;
   765→            this.runPhase = 'INGRESS';
   766→          }
   767→        }
   768→        if (this.hp < this.def.hp * 0.4 && this.airState === 'PATROL') {
   769→          ctx.log(`${this.callsign} · AIRFRAME CRITICAL — RTB`, 'alert');
   770→          this.airState = 'RTB';
   771→        }
   772→        // despawn when far off map
   773→        if (this.x < -300 || this.y < -300 || this.x > ctx.terrain.W + 300 || this.y > ctx.terrain.H + 300) {
   774→          this.airState = 'REARM';
   775→          this.rearmT = 38;
   776→          this.hp = Math.max(this.hp, this.def.hp * 0.55);
   777→          this.damageSmokeT = 0;
   778→        }
   779→        break;
   780→      }
   781→
   782→      case 'RTB': {
   783→        // exit toward SW
   784→        const desired = angleOf(-600 - this.x, 3300 - this.y);
   785→        const diff = angDiff(this.angle, desired);
   786→        this.angle += clamp(diff, -this.def.turnRate * dt, this.def.turnRate * dt);
   787→        this.x += Math.cos(this.angle) * speed * dt;
   788→        this.y += Math.sin(this.angle) * speed * dt;
   789→        this.bank = clamp(this.bank + (Math.sign(diff) - this.bank) * dt * 2, -1, 1);
   790→        if (this.x < -260 || this.y > ctx.terrain.H + 260) {
   791→          this.airState = 'REARM';
   792→          this.rearmT = 34;
   793→          this.hp = Math.max(this.hp, this.def.hp * 0.6);
   794→        }
   795→        break;
   796→      }
   797→
   798→      case 'REARM': {
   799→        this.rearmT -= dt;
   800→        if (this.rearmT <= 0) {
   801→          this.airState = 'INBOUND';
   802→          this.ammo = this.def.ammo;
   803→          this.hp = this.def.hp;
   804→          this.x = 200;
   805→          this.y = ctx.terrain.H - 60;
   806→          this.angle = -Math.PI / 3;
   807→          ctx.log(`${this.callsign} · ON STATION — INBOUND`, 'info');
   808→        }
   809→        break;
   810→      }
   811→
   812→      case 'DOWN':
   813→        break;
   814→    }
   815→
   816→    // aircraft damaged smoke
   817→    if (this.hp < this.def.hp * 0.55 && (this.airState === 'PATROL' || this.airState === 'RTB')) {
   818→      this.damageSmokeT -= dt;
   819→      if (this.damageSmokeT <= 0) {
   820→        this.damageSmokeT = 0.22;
   821→        ctx.effects.spawnSmoke(this.x, this.y, { r: 1.6, r1: 9, life: 2.4, alpha: 0.34 });
   822→      }
   823→    }
   824→
   825→    // separation from other aircraft is unnecessary (only 2, offset patrols)
   826→    void this.turretAngle;
   827→  }

   848→  takeDamage(dmg: number, ctx: SimContext, projKind?: string, attacker?: Unit, aspect = 1) {
   849→    if (this.dead) return;
   850→    let d = dmg * aspect;
   851→    if (projKind === 'AUTO') {
   852→      if (this.def.kind === 'MBT') d *= 0.28;
   853→      else if (this.def.kind === 'IFV') d *= 0.7;
   854→      else if (this.def.kind === 'HQ') d *= 0.12;
   855→      else if (this.def.kind === 'FACTORY') d *= 0.06;
   856→    }
   857→    if (projKind === 'SHELL' && this.def.kind === 'HQ') d *= 0.55;

   859→    if (projKind === 'MISSILE_AIR' && this.def.kind === 'HQ') d *= 0.8;
   860→    if (projKind === 'MISSILE_AIR' && this.def.kind === 'FACTORY') d *= 1.25;
   861→    if (projKind === 'ARTY' && this.def.kind === 'FACTORY') d *= 1.15;

   865→    this.suppression = Math.min(1, this.suppression + 0.12 + Math.min(0.24, d / 140));
   866→    if (attacker) {
   867→      this.lastAttacker = attacker;
   868→      this.lastAttackedT = ctx.time;
   869→    }
   870→    if (this.hp <= 0) {
   871→      this.hp = 0;
   872→      this.die(ctx);
   873→    }
