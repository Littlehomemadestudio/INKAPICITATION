   176→  constructor(seed: number, camera: Camera, audio: AudioEngine, terrainW: number, terrainH: number) {
   177→    this.rng = new RNG((seed ^ 0xEFFEC7) >>> 0);
   178→    this.camera = camera;

   184→    this.scars.height = Math.round(terrainH * SCAR_SCALE);
   185→    this.scarCtx = this.scars.getContext('2d')!;
   186→    for (let i = 0; i < 4; i++) this.smokeSprites.push(this.makeSmokeSprite(seed + i * 977));
   187→  }

   189→  // ── sprite factory ─────────────────────────────────────────

   191→  private makeSmokeSprite(seed: number): HTMLCanvasElement {
   192→    const cv = document.createElement('canvas');
   193→    cv.width = 128;
   194→    cv.height = 128;
   195→    const ctx = cv.getContext('2d')!;
   196→    const rng = new RNG(seed >>> 0);
   197→    for (let i = 0; i < 14; i++) {
   198→      const a = rng.range(0, Math.PI * 2);
   199→      const d = rng.range(0, 34);
   200→      const cx = 64 + Math.cos(a) * d;
   201→      const cy = 64 + Math.sin(a) * d;
   202→      const r = rng.range(20, 44);
   203→      const g = ctx.createRadialGradient(cx, cy, r * 0.1, cx, cy, r);
   204→      g.addColorStop(0, 'rgba(56,52,44,0.34)');
   205→      g.addColorStop(0.7, 'rgba(56,52,44,0.16)');
   206→      g.addColorStop(1, 'rgba(56,52,44,0)');
   207→      ctx.fillStyle = g;
   208→      ctx.beginPath();
   209→      ctx.arc(cx, cy, r, 0, Math.PI * 2);
   210→      ctx.fill();
   211→    }
   212→    return cv;
   213→  }

   217→  spawnExplosion(
   218→    x: number,
   219→    y: number,
   220→    opts: {
   221→      dir?: number;
   222→      dirStrength?: number;
   223→      scale: number;
   224→      crater?: number;
   225→      smoke?: number;
   226→      debris?: number;
   227→      stains?: number;
   228→      ring?: boolean;
   229→      sound?: 'shell' | 'arty' | 'missile' | 'kill' | 'small';
   230→      shake?: number;
   231→    }
   232→  ) {
   233→    const rng = this.rng;
   234→    const dir = opts.dir ?? rng.range(0, Math.PI * 2);
   235→    const dirStrength = opts.dirStrength ?? 0.35;
   236→    const scale = opts.scale;
   237→    const nBlobs = Math.round(4 + scale * rng.range(3.5, 5.5));
   238→    const blobs: InkBlob[] = [];
   239→    for (let i = 0; i < nBlobs; i++) {
   240→      const bias = (rng.next() - 0.5) * Math.PI * dirStrength * 2;
   241→      const a = dir + bias + (rng.next() - 0.5) * Math.PI * 1.5;
   242→      blobs.push({
   243→        a,
   244→        d0: scale * rng.range(1, 4),
   245→        d1: scale * rng.range(8, 22),
   246→        r0: scale * rng.range(1.4, 3.2),
   247→        r1: scale * rng.range(5, 12),
   248→        seed: rng.next() * 1000,
   249→        fade: rng.range(0.55, 1),
   250→      });
   251→    }
   252→    const streaks: Streak[] = [];
   253→    const nStreaks = Math.round(10 + scale * 14);
   254→    for (let i = 0; i < nStreaks; i++) {
   255→      const bias = (rng.next() - 0.5) * Math.PI * dirStrength * 2.4;
   256→      streaks.push({
   257→        a: dir + bias + (rng.next() - 0.5) * Math.PI * 1.7,
   258→        len: scale * rng.range(4, 26),
   259→        w: rng.range(0.5, 1.9),
   260→        curve: rng.range(-0.5, 0.5),
   261→        delay: rng.range(0, 0.09),
   262→      });
   263→    }
   264→    this.explosions.push({
   265→      x,
   266→      y,
   267→      dir,
   268→      dirStrength,
   269→      scale,
   270→      t: 0,
   271→      blobs,
   272→      streaks,
   273→      smokeLeft: opts.smoke ?? Math.round(2 + scale * 3),
   274→      smokeInterval: 0.14,
   275→      smokeTimer: 0,
   276→      ringR: scale * 6,
   277→      hasRing: opts.ring ?? scale > 1.3,
   278→      flash: 1,
   279→    });
   280→
   281→    // debris
   282→    const nDebris = opts.debris ?? Math.round(scale * rng.range(4, 9));
   283→    for (let i = 0; i < nDebris; i++) {
   284→      const a = dir + (rng.next() - 0.5) * Math.PI * 1.9;
   285→      const sp = scale * rng.range(14, 52);
   286→      this.debris.push({
   287→        x,
   288→        y,
   289→        z: rng.range(2, 9),
   290→        vx: Math.cos(a) * sp,
   291→        vy: Math.sin(a) * sp,
   292→        vz: rng.range(18, 46) * Math.sqrt(scale),
   293→        spin: rng.range(-9, 9),
   294→        rot: rng.range(0, Math.PI * 2),
   295→        size: rng.range(0.5, 1.4) * Math.sqrt(scale),
   296→        life: 0,
   297→        landed: false,
   298→        isTurret: false,
   299→        turretAngle: 0,
   300→      });
   301→    }
   302→
   303→    // permanent scarring
   304→    if (opts.crater && opts.crater > 0) this.stampCrater(x, y, opts.crater);
   305→    const nStains = opts.stains ?? Math.round(scale * rng.range(14, 26));
   306→    if (nStains > 0) this.stampStains(x, y, dir, dirStrength, scale, nStains);
   307→
   308→    if (opts.ring) {
   309→      this.rings.push({ x, y, r: scale * 4, r1: scale * 34, age: 0, life: 0.7 });
   310→    }
   311→
   312→    // audio + shake
   313→    if (opts.sound) this.audio.explosion(opts.sound, x, y, scale);
   314→    const sh = opts.shake ?? scale * 2.2;
   315→    if (sh > 0) this.camera.addShake(x, y, sh);
   316→  }

   361→  /** schedule a train of secondary explosions — structural collapse */
   362→  scheduleBlasts(
   363→    x: number,
   364→    y: number,
   365→    opts: { count: number; duration: number; spread: number; scaleMin: number; scaleMax: number; delay?: number }
   366→  ) {
   367→    const delay = opts.delay ?? 0;
   368→    for (let i = 0; i < opts.count; i++) {
   369→      const a = this.rng.range(0, Math.PI * 2);
   370→      const d = this.rng.range(0, opts.spread);
   371→      this.pending.push({
   372→        x: x + Math.cos(a) * d,
   373→        y: y + Math.sin(a) * d,
   374→        at: this.time + delay + (opts.duration * (i + this.rng.range(0, 0.6))) / opts.count,
   375→        scale: this.rng.range(opts.scaleMin, opts.scaleMax),
   376→        crater: this.rng.range(4, 9),
   377→        smoke: 7,
   378→        debris: 9,
   379→        stains: 22,
   380→        ring: true,
   381→        shake: 3,
   382→        sound: 'arty',
   383→      });
   384→    }
   385→  }

   389→  stampCrater(x: number, y: number, r: number) {
   390→    this.craters.push({ x, y, r, seed: this.rng.next() * 100 });
   391→    if (this.craters.length > 300) this.craters.shift();
   392→    const ctx = this.scarCtx;
   393→    ctx.save();
   394→    ctx.translate(x * SCAR_SCALE, y * SCAR_SCALE);
   395→    ctx.scale(SCAR_SCALE, SCAR_SCALE);
   396→    // irregular hole
   397→    ctx.fillStyle = 'rgba(24,21,14,0.52)';
   398→    ctx.beginPath();
   399→    const n = 11;
   400→    for (let i = 0; i <= n; i++) {
   401→      const a = (i / n) * Math.PI * 2;
   402→      const rr = r * (0.72 + 0.38 * Math.abs(Math.sin(i * 3.1 + r)));
   403→      const px = Math.cos(a) * rr;
   404→      const py = Math.sin(a) * rr * 0.88;
   405→      if (i === 0) ctx.moveTo(px, py);
   406→      else ctx.lineTo(px, py);
   407→    }
   408→    ctx.closePath();
   409→    ctx.fill();
   410→    // darker heart
   411→    ctx.fillStyle = 'rgba(12,10,6,0.5)';
   412→    ctx.beginPath();
   413→    ctx.ellipse(0, 0, r * 0.42, r * 0.36, 0.3, 0, Math.PI * 2);
   414→    ctx.fill();
   415→    // NW rim light
   416→    ctx.strokeStyle = 'rgba(255,252,240,0.4)';
   417→    ctx.lineWidth = r * 0.16;
   418→    ctx.beginPath();
   419→    ctx.arc(0, 0, r * 0.9, Math.PI * 0.85, Math.PI * 1.6);
   420→    ctx.stroke();
   421→    // radial cracks
   422→    ctx.strokeStyle = 'rgba(20,17,11,0.4)';
   423→    ctx.lineWidth = r * 0.07;
   424→    ctx.beginPath();
   425→    for (let i = 0; i < 5; i++) {
   426→      const a = this.rng.range(0, Math.PI * 2);
   427→      const l = r * this.rng.range(1.3, 2.4);
   428→      ctx.moveTo(Math.cos(a) * r * 0.7, Math.sin(a) * r * 0.7);
   429→      ctx.lineTo(Math.cos(a) * l, Math.sin(a) * l);
   430→    }
   431→    ctx.stroke();
   432→    ctx.restore();
   433→  }

   435→  stampStains(x: number, y: number, dir: number, dirStrength: number, scale: number, count: number) {
   436→    const ctx = this.scarCtx;
   437→    ctx.save();
   438→    ctx.translate(x * SCAR_SCALE, y * SCAR_SCALE);
   439→    ctx.scale(SCAR_SCALE, SCAR_SCALE);
   440→    for (let i = 0; i < count; i++) {
   441→      const bias = (this.rng.next() - 0.5) * Math.PI * (0.5 + dirStrength);
   442→      const a = dir + bias + (this.rng.next() - 0.5) * Math.PI * 1.6;
   443→      const d = scale * this.rng.range(3, 26);
   444→      const px = Math.cos(a) * d;
   445→      const py = Math.sin(a) * d;
   446→      const r = this.rng.range(0.4, 1.1) * (0.7 + scale * 0.5);
   447→      const elong = this.rng.range(1, 3.2);
   448→      ctx.fillStyle = `rgba(16,13,8,${this.rng.range(0.25, 0.6)})`;
   449→      ctx.beginPath();
   450→      ctx.ellipse(px, py, r * elong, r, a, 0, Math.PI * 2);
   451→      ctx.fill();
   452→    }
   453→    ctx.restore();
   454→  }

   465→    ctx.beginPath();
   466→    ctx.arc(0, 0, r, 0, Math.PI * 2);

   484→  update(dt: number) {
   485→    this.time += dt;
   486→    this.windAngle = -0.6 + Math.sin(this.time * 0.013) * 0.45;
   487→
   488→    // scheduled secondary blasts
   489→    for (let i = this.pending.length - 1; i >= 0; i--) {
   490→      const b = this.pending[i];
   491→      if (this.time >= b.at) {
   492→        this.spawnExplosion(b.x, b.y, {
   493→          dir: this.rng.range(0, Math.PI * 2),
   494→          dirStrength: 0.4,
   495→          scale: b.scale,
   496→          crater: b.crater,
   497→          smoke: b.smoke,
   498→          debris: b.debris,
   499→          stains: b.stains,
   500→          ring: b.ring,
   501→          sound: b.sound,
   502→          shake: b.shake,
   503→        });
   504→        this.pending.splice(i, 1);
   505→      }
   506→    }
   507→
   508→    // explosions
   509→    for (let i = this.explosions.length - 1; i >= 0; i--) {
   510→      const e = this.explosions[i];
   511→      e.t += dt;
   512→      e.flash = Math.max(0, e.flash - dt * 14);
   513→      // staged smoke
   514→      if (e.smokeLeft > 0) {
   515→        e.smokeTimer -= dt;
   516→        if (e.smokeTimer <= 0) {
   517→          e.smokeTimer = e.smokeInterval;
   518→          e.smokeLeft--;
   519→          const jitter = e.scale * 3;
   520→          const early = e.t < 0.5;
   521→          this.spawnSmoke(
   522→            e.x + this.rng.range(-jitter, jitter),
   523→            e.y + this.rng.range(-jitter, jitter),
   524→            {
   525→              r: e.scale * this.rng.range(2.5, 5),
   526→              r1: e.scale * this.rng.range(18, 34) + 14,
   527→              life: this.rng.range(3.4, 7.5),
   528→              alpha: early ? this.rng.range(0.5, 0.62) : this.rng.range(0.34, 0.5),
   529→              vx: Math.cos(e.dir) * e.scale * 3 * e.dirStrength,
   530→              vy: Math.sin(e.dir) * e.scale * 3 * e.dirStrength,
   531→              dark: early ? 1.6 : 1,
   532→            }
   533→          );
   534→        }
   535→      }
   536→      if (e.t > 1.4) this.explosions.splice(i, 1);
   537→    }
   538→
   539→    // debris
   540→    for (let i = this.debris.length - 1; i >= 0; i--) {
   541→      const d = this.debris[i];
   542→      d.life += dt;
   543→      if (d.landed) continue;
   544→      d.x += d.vx * dt;
   545→      d.y += d.vy * dt;
   546→      d.z += d.vz * dt;
   547→      d.vz -= 90 * dt;
   548→      d.rot += d.spin * dt;
   549→      if (d.z <= 0) {
   550→        d.z = 0;
   551→        d.landed = true;
   552→        if (d.isTurret) {
   553→          this.stampScorch(d.x, d.y, 7, 0.3);
   554→        } else {
   555→          this.stampScorch(d.x, d.y, 1.4 * d.size, 0.22);
   556→        }
   557→        d.vx = 0;
   558→        d.vy = 0;
   559→      }
   560→      if (d.landed && d.life > 0) {
   561→        // keep landed debris visible for a while, then bake tiny stain
   562→        if (d.life > 6 && !d.isTurret) this.debris.splice(i, 1);
   563→      }
   564→    }
   565→
   566→    // smoke
   567→    const wx = Math.cos(this.windAngle) * this.windSpeed;
   568→    const wy = Math.sin(this.windAngle) * this.windSpeed;
   569→    for (let i = this.smokes.length - 1; i >= 0; i--) {
   570→      const s = this.smokes[i];
   571→      s.age += dt;
   572→      if (s.age >= s.life) {
   573→        this.smokes.splice(i, 1);
   574→        continue;
   575→      }
   576→      s.x += (s.vx + wx * 0.4) * dt;
   577→      s.y += (s.vy + wy * 0.4) * dt;
   578→      s.vx *= 1 - dt * 0.4;
   579→      s.vy *= 1 - dt * 0.4;
   580→      s.rot += dt * 0.1;
   581→    }
   582→
   583→    // rings & flashes
   584→    for (let i = this.rings.length - 1; i >= 0; i--) {
   585→      const r = this.rings[i];
   586→      r.age += dt;
   587→      if (r.age >= r.life) this.rings.splice(i, 1);
   588→    }
   589→    for (let i = this.flashes.length - 1; i >= 0; i--) {
   590→      this.flashes[i].age += dt;
   591→      if (this.flashes[i].age > 0.09) this.flashes.splice(i, 1);
   592→    }
   593→
   594→    // order markers
   595→    for (let i = this.orderMarkers.length - 1; i >= 0; i--) {
   596→      this.orderMarkers[i].t += dt;
   597→      if (this.orderMarkers[i].t > 0.8) this.orderMarkers.splice(i, 1);
   598→    }
   599→
   600→    // wreck smoke
   601→    for (const w of this.wrecks) {
   602→      if (this.time < w.smokeUntil && this.smokes.length < 380) {
   603→        if (this.rng.chance(dt * 2.2)) {
   604→          const life = clamp((w.smokeUntil - this.time) / 20, 0.4, 3.6);
   605→          this.spawnSmoke(w.x + this.rng.range(-3, 3), w.y + this.rng.range(-3, 3), {
   606→            r: 2.5,
   607→            r1: 15 + this.rng.range(0, 14),
   608→            life: 2.5 + this.rng.range(0, 3),
   609→            alpha: 0.24 + 0.14 * clamp((w.smokeUntil - this.time) / 40, 0, 1),
   610→            vy: -3,
   611→          });
   612→          void life;
   613→        }
   614→      }
   615→    }
   616→
   617→    // rubble smoke — long oily columns off a destroyed structure
   618→    for (const r of this.rubble) {
   619→      if (this.time < r.smokeUntil && this.smokes.length < 380) {
   620→        if (this.rng.chance(dt * 1.6)) {
   621→          this.spawnSmoke(
   622→            r.x + this.rng.range(-r.w * 0.3, r.w * 0.3),
   623→            r.y + this.rng.range(-r.h * 0.3, r.h * 0.3),
   624→            {
   625→              r: 3,
   626→              r1: 20 + this.rng.range(0, 18),
   627→              life: 4 + this.rng.range(0, 4),
   628→              alpha: 0.3,
   629→              vy: -5,
   630→              dark: 1.3,
   631→            }
   632→          );
   633→        }
   634→      }
   635→    }
   636→  }

   698→  /** ink cores, debris, rings, flashes — above units */
   699→  drawCore(ctx: CanvasRenderingContext2D) {
   700→    // explosion ink
   701→    for (const e of this.explosions) {
   702→      const t = e.t;
   703→      const grow = clamp(t / 0.28, 0, 1);
   704→      const fade = clamp(1 - (t - 0.35) / 0.9, 0, 1);
   705→      const ink = clamp(fade, 0, 1);
   706→      ctx.fillStyle = `rgba(16,13,8,${0.8 * ink})`;
   707→      for (const b of e.blobs) {
   708→        const d = b.d0 + (b.d1 - b.d0) * grow;
   709→        const r = b.r0 + (b.r1 - b.r0) * grow;
   710→        const bx = e.x + Math.cos(b.a) * d;
   711→        const by = e.y + Math.sin(b.a) * d;
   712→        // irregular ink blob
   713→        ctx.beginPath();
   714→        const n = 9;
   715→        for (let i = 0; i <= n; i++) {
   716→          const a = (i / n) * Math.PI * 2 + b.seed;
   717→          const rr = r * (0.6 + 0.45 * Math.abs(Math.sin(i * 2.3 + b.seed)));
   718→          const px = bx + Math.cos(a) * rr;
   719→          const py = by + Math.sin(a) * rr;
   720→          if (i === 0) ctx.moveTo(px, py);
   721→          else ctx.lineTo(px, py);
   722→        }
   723→        ctx.closePath();
   724→        ctx.globalAlpha = b.fade * ink;
   725→        ctx.fill();
   726→      }
   727→      ctx.globalAlpha = 1;
   728→      // spatter streaks
   729→      ctx.strokeStyle = `rgba(14,11,7,${0.75 * ink})`;
   730→      ctx.lineCap = 'round';
   731→      for (const s of e.streaks) {
   732→        const tt = clamp((t - s.delay) / 0.2, 0, 1);
   733→        if (tt <= 0) continue;
   734→        const d0 = e.scale * 2 + tt * s.len;
   735→        const d1 = d0 + s.len * 0.5 * tt;
   736→        ctx.lineWidth = s.w * (1 - tt * 0.5);
   737→        ctx.beginPath();
   738→        ctx.moveTo(e.x + Math.cos(s.a) * d0, e.y + Math.sin(s.a) * d0);
   739→        // curved streak
   740→        const midA = s.a + s.curve * 0.35;
   741→        ctx.quadraticCurveTo(
   742→          e.x + Math.cos(midA) * (d0 + d1) * 0.5,
   743→          e.y + Math.sin(midA) * (d0 + d1) * 0.5,
   744→          e.x + Math.cos(s.a) * d1,
   745→          e.y + Math.sin(s.a) * d1
   746→        );
   747→        ctx.stroke();
   748→      }
   749→      // white-hot flash core (brief)
   750→      if (e.flash > 0) {
   751→        ctx.fillStyle = `rgba(255,253,246,${e.flash * 0.9})`;
   752→        ctx.beginPath();
   753→        ctx.arc(e.x, e.y, e.scale * 2.6 * (1.2 - e.flash * 0.4), 0, Math.PI * 2);
   754→        ctx.fill();
   755→        // directional hot wedges
   756→        ctx.fillStyle = `rgba(255,253,246,${e.flash * 0.55})`;
   757→        for (let i = 0; i < 4; i++) {
   758→          const a = e.dir + (i - 1.5) * 0.4;
   759→          ctx.beginPath();
   760→          ctx.moveTo(e.x, e.y);
   761→          ctx.lineTo(e.x + Math.cos(a) * e.scale * 8 * e.flash, e.y + Math.sin(a) * e.scale * 8 * e.flash);
   762→          ctx.lineTo(e.x + Math.cos(a + 0.22) * e.scale * 4 * e.flash, e.y + Math.sin(a + 0.22) * e.scale * 4 * e.flash);
   763→          ctx.closePath();
   764→          ctx.fill();
   765→        }
   766→      }
   767→    }
   768→
   769→    // debris
   770→    for (const d of this.debris) {
   771→      if (d.isTurret) continue; // drawn in wreck layer
   772→      const px = d.x;
   773→      const py = d.y - d.z * 0.35;
   774→      // ground shadow
   775→      ctx.fillStyle = 'rgba(20,17,11,0.18)';
   776→      ctx.beginPath();
   777→      ctx.arc(d.x, d.y, d.size * 0.9, 0, Math.PI * 2);
   778→      ctx.fill();
   779→      ctx.save();
   780→      ctx.translate(px, py);
   781→      ctx.rotate(d.rot);
   782→      ctx.fillStyle = d.landed ? 'rgba(24,20,14,0.8)' : '#171410';
   783→      ctx.fillRect(-d.size, -d.size * 0.55, d.size * 2, d.size * 1.1);
   784→      ctx.restore();
   785→    }
   786→
   787→    // shockwave rings
   788→    for (const r of this.rings) {
   789→      const t = r.age / r.life;
   790→      const rr = r.r + (r.r1 - r.r) * t;
   791→      ctx.strokeStyle = `rgba(30,26,18,${0.22 * (1 - t)})`;
   792→      ctx.lineWidth = 1.4;
   793→      ctx.beginPath();
   794→      ctx.arc(r.x, r.y, rr, 0, Math.PI * 2);
   795→      ctx.stroke();
   796→    }
   797→
   798→    // muzzle flashes
   799→    for (const f of this.flashes) {
   800→      const a = f.age / 0.09;
   801→      const alpha = 1 - a;
   802→      ctx.save();
   803→      ctx.translate(f.x, f.y);
   804→      ctx.rotate(f.angle);
   805→      // dark propellant wash
   806→      ctx.fillStyle = `rgba(30,26,18,${0.35 * alpha})`;
   807→      ctx.beginPath();
   808→      ctx.moveTo(0, -f.size * 1.4);
   809→      ctx.lineTo(f.size * 7, 0);
   810→      ctx.lineTo(0, f.size * 1.4);
   811→      ctx.closePath();
   812→      ctx.fill();
   813→      // hot core
   814→      ctx.fillStyle = `rgba(255,254,248,${0.85 * alpha})`;
   815→      ctx.beginPath();
   816→      ctx.arc(0, 0, f.size * 1.1, 0, Math.PI * 2);
   817→      ctx.fill();
   818→      ctx.restore();
   819→    }
   820→  }
