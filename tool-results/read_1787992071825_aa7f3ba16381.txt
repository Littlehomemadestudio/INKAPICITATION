   191→   191→  // ── height ─────────────────────────────────────────────────
   192→   192→  //
   193→   193→  // Hand-composed relief — the map is authored, not noise soup:
   194→   194→  //   · ZAPAD RIDGE  — elongated NE–SW ridgeline in the west
   195→   195→  //   · HILL 214     — dominant high ground, east-centre
   196→   196→  //   · HILL 163     — detached knoll north of the river
   197→   197→  //   · southern rolling rise screening the player's entry
   198→   198→  //   · NE plateau carrying the enemy HQ
   199→   199→  //   · river valley carve + dry tributary
   200→   200→  //
   201→   201→  heightAt(x: number, y: number): number {
   202→   202→    let h = this.baseNoise.fbm(x / 1500, y / 1500, 5) * 26 - 6;
   203→   203→    h += this.detailNoise.fbm(x / 380, y / 380, 3) * 5 - 1.5;
   204→   204→
   205→   205→    // ZAPAD RIDGE — four overlapping crests along one spine
   206→   206→    h += this.gaussEl(x, y, 500, 480, 380, 300, 28);
   207→   207→    h += this.gaussEl(x, y, 730, 660, 420, 320, 36);
   208→   208→    h += this.gaussEl(x, y, 950, 840, 400, 300, 30);
   209→   209→    h += this.gaussEl(x, y, 1140, 1010, 360, 280, 22);
   210→   210→
   211→   211→    // HILL 214 — the dominant landform
   212→   212→    h += this.gauss(x, y, 2950, 1180, 620, 58);
   213→   213→    // shoulder south-west of the hill (finger toward the village)
   214→   214→    h += this.gaussEl(x, y, 2620, 1420, 360, 260, 14);
   215→   215→
   216→   216→    // HILL 163 knoll
   217→   217→    h += this.gauss(x, y, 1550, 680, 360, 26);
   218→   218→
   219→   219→    // southern rise near player entry
   220→   220→    h += this.gauss(x, y, 1750, 2750, 520, 16);
   221→   221→    // south-eastern rolling ground
   222→   222→    h += this.gaussEl(x, y, 3350, 2600, 600, 380, 12);
   223→   223→
   224→   224→    // north-east plateau toward HQ
   225→   225→    const nx = clamp(x / this.W, 0, 1);
   226→   226→    const ny = 1 - clamp(y / this.H, 0, 1);
   227→   227→    h += smooth01(nx * 0.7 + ny * 0.7) * 22;
   228→   228→
   229→   229→    // river valley carve
   230→   230→    const dRiver = this.distToPolyline(x, y, this.river);
   231→   231→    if (dRiver < 170) {
   232→   232→      const t = 1 - dRiver / 170;
   233→   233→      h -= smooth01(t) * 11;
   234→   234→    }
   235→   235→    // islands rise out of the bay — real ground for LOS and contours
   236→   236→    for (const isl of this.sea.islands) {
   237→   237→      h += this.gaussEl(x, y, isl.x, isl.y, isl.rx * 1.35, isl.ry * 1.35, isl.height);
   238→   238→    }
   239→   239→    // the sea floor is flat: the water surface carries the depth story
   240→   240→    if (this.sea.isSea(x, y)) return 0;
   241→   241→    return Math.max(0, h);
   242→   242→  }
   243→   243→
   244→   244→  slopeAt(x: number, y: number): number {
   245→   245→    const d = 24;
   246→   246→    const hx = this.heightAt(x + d, y) - this.heightAt(x - d, y);
   247→   247→    const hy = this.heightAt(x, y + d) - this.heightAt(x, y - d);
   248→   248→    return Math.hypot(hx, hy) / (2 * d);
   249→   249→  }
   250→   250→
   251→   251→  private gauss(x: number, y: number, cx: number, cy: number, r: number, amp: number): number {
   252→   252→    const d = dist(x, y, cx, cy);
   253→   253→    if (d > r) return 0;
   254→   254→    const t = 1 - d / r;
   255→   255→    return amp * t * t * (3 - 2 * t);
   256→   256→  }
   257→   257→
   258→   258→  /** elongated gaussian — ridge segments */
   259→   259→  private gaussEl(x: number, y: number, cx: number, cy: number, rx: number, ry: number, amp: number): number {
   260→   260→    const dx = (x - cx) / rx;
   261→   261→    const dy = (y - cy) / ry;
   262→   262→    const d2 = dx * dx + dy * dy;
   263→   263→    if (d2 > 1) return 0;
   264→   264→    const t = 1 - Math.sqrt(d2);
   265→   265→    return amp * t * t * (3 - 2 * t);
   266→   266→  }

   268→   268→  // ── LOS over the height grid ───────────────────────────────
   269→   269→
   270→   270→  /** terrain line of sight between two points at given eye heights;
   271→   271→   *  buildings interrupt observation and direct fire */
   272→   272→  losClear(ax: number, ay: number, aEye: number, bx: number, by: number, bEye: number, includeBuildings = true): boolean {
   273→   273→    const d = dist(ax, ay, bx, by);
   274→   274→    const steps = Math.max(2, Math.ceil(d / 56));
   275→   275→    for (let s = 1; s < steps; s++) {
   276→   276→      const t = s / steps;
   277→   277→      const x = ax + (bx - ax) * t;
   278→   278→      const y = ay + (by - ay) * t;
   279→   279→      const beam = aEye + (bEye - aEye) * t;
   280→   280→      const ground = this.gridHeight(x, y);
   281→   281→      if (ground > beam + 2.0) return false;
   282→   282→    }
   283→   283→    if (includeBuildings) {
   284→   284→      // sample the beam against building footprints (they are tall enough
   285→   285→      // to matter to a ground-level observer; aircraft pass above)
   286→   286→      if (aEye < 30 && bEye < 30) {
   287→   287→        const bsteps = Math.max(2, Math.ceil(d / 16));
   288→   288→        for (let s = 1; s < bsteps; s++) {
   289→   289→          const t = s / bsteps;
   290→   290→          const x = ax + (bx - ax) * t;
   291→   291→          const y = ay + (by - ay) * t;
   292→   292→          const gx = (x / this.BM_CELL) | 0;
   293→   293→          const gy = (y / this.BM_CELL) | 0;
   294→   294→          if (gx < 0 || gy < 0 || gx >= this.bmw || gy >= this.bmh) continue;
   295→   295→          const idx = gy * this.bmw + gx;
   296→   296→          if (this.bmask[idx] || this.rockMask[idx]) return false;
   297→   297→        }
   298→   298→      }
   299→   299→    }
   300→   300→    return true;
   301→   301→  }
   302→   302→
   303→   303→  private gridHeight(x: number, y: number): number {
   304→   304→    const gx = clamp(x / HG_STEP, 0, this.hgw - 1.001);
   305→   305→    const gy = clamp(y / HG_STEP, 0, this.hgh - 1.001);
   306→   306→    const ix = Math.floor(gx);
   307→   307→    const iy = Math.floor(gy);
   308→   308→    const fx = gx - ix;
   309→   309→    const fy = gy - iy;
   310→   310→    const h00 = this.hg[iy * this.hgw + ix];
   311→   311→    const h10 = this.hg[iy * this.hgw + ix + 1];
   312→   312→    const h01 = this.hg[(iy + 1) * this.hgw + ix];
   313→   313→    const h11 = this.hg[(iy + 1) * this.hgw + ix + 1];
   314→   314→    return (h00 * (1 - fx) + h10 * fx) * (1 - fy) + (h01 * (1 - fx) + h11 * fx) * fy;
   315→   315→  }

   367→   367→  forestDensity(x: number, y: number): number {
   368→   368→    const n = this.forestNoise.fbm(x / 520, y / 520, 4);
