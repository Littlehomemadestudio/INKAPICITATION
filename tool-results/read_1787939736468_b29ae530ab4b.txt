   176→  constructor(seed: number) {
   177→    this.seed = seed >>> 0;
   178→    this.baseNoise = new Noise2D(this.seed);
   179→    this.detailNoise = new Noise2D((this.seed ^ 0x9e3779b9) >>> 0);

   183→
   184→  // ── height ─────────────────────────────────────────────────
   185→  //
   186→  // Hand-composed relief — the map is authored, not noise soup:
   187→  //   · ZAPAD RIDGE  — elongated NE–SW ridgeline in the west
   188→  //   · HILL 214     — dominant high ground, east-centre
   189→  //   · HILL 163     — detached knoll north of the river
   190→  //   · southern rolling rise screening the player's entry
   191→  //   · NE plateau carrying the enemy HQ
   192→  //   · river valley carve + dry tributary
   193→  //
   194→  heightAt(x: number, y: number): number {
   195→    let h = this.baseNoise.fbm(x / 1500, y / 1500, 5) * 26 - 6;
   196→    h += this.detailNoise.fbm(x / 380, y / 380, 3) * 5 - 1.5;
   197→
   198→    // ZAPAD RIDGE — four overlapping crests along one spine
   199→    h += this.gaussEl(x, y, 500, 480, 380, 300, 28);
   200→    h += this.gaussEl(x, y, 730, 660, 420, 320, 36);
   201→    h += this.gaussEl(x, y, 950, 840, 400, 300, 30);
   202→    h += this.gaussEl(x, y, 1140, 1010, 360, 280, 22);
   203→
   204→    // HILL 214 — the dominant landform
   205→    h += this.gauss(x, y, 2950, 1180, 620, 58);
   206→    // shoulder south-west of the hill (finger toward the village)
   207→    h += this.gaussEl(x, y, 2620, 1420, 360, 260, 14);
   208→
   209→    // HILL 163 knoll
   210→    h += this.gauss(x, y, 1550, 680, 360, 26);
   211→
   212→    // southern rise near player entry
   213→    h += this.gauss(x, y, 1750, 2750, 520, 16);
   214→    // south-eastern rolling ground
   215→    h += this.gaussEl(x, y, 3350, 2600, 600, 380, 12);
   216→
   217→    // north-east plateau toward HQ
   218→    const nx = clamp(x / this.W, 0, 1);
   219→    const ny = 1 - clamp(y / this.H, 0, 1);
   220→    h += smooth01(nx * 0.7 + ny * 0.7) * 22;
   221→
   222→    // river valley carve
   223→    const dRiver = this.distToPolyline(x, y, this.river);
   224→    if (dRiver < 170) {
   225→      const t = 1 - dRiver / 170;
   226→      h -= smooth01(t) * 11;
   227→    }
   228→    return Math.max(0, h);
   229→  }
   230→
   231→  slopeAt(x: number, y: number): number {
   232→    const d = 24;
   233→    const hx = this.heightAt(x + d, y) - this.heightAt(x - d, y);
   234→    const hy = this.heightAt(x, y + d) - this.heightAt(x, y - d);
   235→    return Math.hypot(hx, hy) / (2 * d);
   236→  }
   237→
   238→  private gauss(x: number, y: number, cx: number, cy: number, r: number, amp: number): number {
   239→    const d = dist(x, y, cx, cy);
   240→    if (d > r) return 0;
   241→    const t = 1 - d / r;
   242→    return amp * t * t * (3 - 2 * t);
   243→  }
   244→
   245→  /** elongated gaussian — ridge segments */
   246→  private gaussEl(x: number, y: number, cx: number, cy: number, rx: number, ry: number, amp: number): number {
   247→    const dx = (x - cx) / rx;
   248→    const dy = (y - cy) / ry;
   249→    const d2 = dx * dx + dy * dy;
   250→    if (d2 > 1) return 0;
   251→    const t = 1 - Math.sqrt(d2);
   252→    return amp * t * t * (3 - 2 * t);
   253→  }

   255→  // ── LOS over the height grid ───────────────────────────────
   256→
   257→  /** terrain line of sight between two points at given eye heights;
   258→   *  buildings interrupt observation and direct fire */
   259→  losClear(ax: number, ay: number, aEye: number, bx: number, by: number, bEye: number, includeBuildings = true): boolean {
   260→    const d = dist(ax, ay, bx, by);
   261→    const steps = Math.max(2, Math.ceil(d / 56));
   262→    for (let s = 1; s < steps; s++) {
   263→      const t = s / steps;
   264→      const x = ax + (bx - ax) * t;
   265→      const y = ay + (by - ay) * t;
   266→      const beam = aEye + (bEye - aEye) * t;
   267→      const ground = this.gridHeight(x, y);
   268→      if (ground > beam + 2.0) return false;
   269→    }
   270→    if (includeBuildings) {
   271→      // sample the beam against building footprints (they are tall enough
   272→      // to matter to a ground-level observer; aircraft pass above)
   273→      if (aEye < 30 && bEye < 30) {
   274→        const bsteps = Math.max(2, Math.ceil(d / 16));
   275→        for (let s = 1; s < bsteps; s++) {
   276→          const t = s / bsteps;
   277→          const x = ax + (bx - ax) * t;
   278→          const y = ay + (by - ay) * t;
   279→          const gx = (x / this.BM_CELL) | 0;
   280→          const gy = (y / this.BM_CELL) | 0;
   281→          if (gx < 0 || gy < 0 || gx >= this.bmw || gy >= this.bmh) continue;
   282→          const idx = gy * this.bmw + gx;
   283→          if (this.bmask[idx] || this.rockMask[idx]) return false;
   284→        }
   285→      }
   286→    }
   287→    return true;
   288→  }
   289→
   290→  private gridHeight(x: number, y: number): number {
   291→    const gx = clamp(x / HG_STEP, 0, this.hgw - 1.001);
   292→    const gy = clamp(y / HG_STEP, 0, this.hgh - 1.001);
   293→    const ix = Math.floor(gx);
   294→    const iy = Math.floor(gy);
   295→    const fx = gx - ix;
   296→    const fy = gy - iy;
   297→    const h00 = this.hg[iy * this.hgw + ix];
   298→    const h10 = this.hg[iy * this.hgw + ix + 1];
   299→    const h01 = this.hg[(iy + 1) * this.hgw + ix];
   300→    const h11 = this.hg[(iy + 1) * this.hgw + ix + 1];
   301→    return (h00 * (1 - fx) + h10 * fx) * (1 - fy) + (h01 * (1 - fx) + h11 * fx) * fy;
   302→  }

   312→  /** shallow ford — vehicles may wade across, slowly */
   313→  fordAt(x: number, y: number, r = 46): boolean {
   314→    if (!this.ford) return false;
   315→    return dist(x, y, this.ford.x, this.ford.y) < r;
   316→  }

   318→  /** bridge within radius */
   319→  bridgeAt(x: number, y: number, r = 42): Bridge | null {
   320→    for (const b of this.bridges) {
   321→      if (dist(x, y, b.x, b.y) < r) return b;
   322→    }
   323→    return null;
   324→  }

   326→  // ── roads & forests ────────────────────────────────────────

   328→  roadFactor(x: number, y: number): number {
   329→    // 0 = no road, 1 = on major road centreline (approx)
   330→    let best = 0;
   331→    for (const r of this.roads) {
   332→      const d = this.distToPolyline(x, y, r.pts);
   333→      const w = r.major ? 26 : 20;
   334→      if (d < w) best = Math.max(best, (r.major ? 1 : 0.7) * (1 - d / w * 0.4));
   335→    }
   336→    return best;
   337→  }
   338→
   339→  railFactor(x: number, y: number): number {
   340→    const d = this.distToPolyline(x, y, this.railway);
   341→    return d < 14 ? 1 - d / 14 * 0.5 : 0;
   342→  }
   343→
   344→  forestDensity(x: number, y: number): number {
   345→    const n = this.forestNoise.fbm(x / 520, y / 520, 4);
   346→    // NW-biased woodland belt; the valley slopes carry brush
   347→    const west = 1 - x / this.W;
   348→    const bias = -0.075 * west + 0.035 * (y / this.H);
   349→    const v = (n - 0.585 + bias) * 4.2;
   350→    return clamp(v, 0, 1) * 1.6;
   351→  }

   806→  /** true if a w×h footprint at (x,y,rot) keeps `margin` m clear of every standing building */
   807→  private buildingFits(x: number, y: number, w: number, h: number, rot: number, margin = 4): boolean {
   808→    const reach = Math.hypot(w, h) / 2;
   809→    for (const b of this.buildings) {
   810→      const br = Math.hypot(b.w, b.h) / 2;
   811→      if (Math.abs(b.x - x) > reach + br + margin || Math.abs(b.y - y) > reach + br + margin) continue;
   812→      // rotated-box SAT — minimum separation across the four axes
   813→      const ca1 = Math.cos(rot), sa1 = Math.sin(rot);
   814→      const ca2 = Math.cos(b.rot), sa2 = Math.sin(b.rot);
   815→      const dx = b.x - x, dy = b.y - y;
   816→      const axes: Array<[number, number]> = [[ca1, sa1], [-sa1, ca1], [ca2, sa2], [-sa2, ca2]];
   817→      let minSep = Infinity;
   818→      for (const [ux, uy] of axes) {
   819→        const dist = dx * ux + dy * uy;
   820→        const rA = (w / 2) * Math.abs(ca1 * ux + sa1 * uy) + (h / 2) * Math.abs(-sa1 * ux + ca1 * uy);
   821→        const rB = (b.w / 2) * Math.abs(ca2 * ux + sa2 * uy) + (b.h / 2) * Math.abs(-sa2 * ux + ca2 * uy);
   822→        minSep = Math.min(minSep, rA + rB - Math.abs(dist));
   823→      }
   824→      if (minSep > -margin) return false;
   825→    }
   826→    return true;
   827→  }

   836→      kinds: BuildingKind[]

   846→        for (let attempt = 0; attempt < 3 && !placed; attempt++) {
   847→          const set = 52 + attempt * 16 + rng.range(-6, 6);

  1001→  /** a collapsed building opens sightlines and routes; rubble remains */
  1002→  onBuildingDestroyed(b: Building) {
  1003→    // clear LOS occupancy over the footprint (intersect-based, matching
  1004→    // the rasterisation so no stale cells survive)
  1005→    const reach = Math.max(b.w, b.h) * 0.5;
  1006→    const gx0 = Math.max(0, ((b.x - reach) / this.BM_CELL) | 0);
  1007→    const gx1 = Math.min(this.bmw - 1, ((b.x + reach) / this.BM_CELL) | 0);
  1008→    const gy0 = Math.max(0, ((b.y - reach) / this.BM_CELL) | 0);
  1009→    const gy1 = Math.min(this.bmh - 1, ((b.y + reach) / this.BM_CELL) | 0);
  1010→    const c = Math.cos(b.rot);
  1011→    const s = Math.sin(b.rot);
  1012→    for (let gy = gy0; gy <= gy1; gy++) {
  1013→      for (let gx = gx0; gx <= gx1; gx++) {
  1014→        const px = clamp(b.x, gx * this.BM_CELL, (gx + 1) * this.BM_CELL);
  1015→        const py = clamp(b.y, gy * this.BM_CELL, (gy + 1) * this.BM_CELL);
  1016→        const dx = px - b.x;
  1017→        const dy = py - b.y;
  1018→        const lx = dx * c + dy * s;
  1019→        const ly = -dx * s + dy * c;
  1020→        if (Math.abs(lx) <= b.w / 2 && Math.abs(ly) <= b.h / 2) {
  1021→          this.bmask[gy * this.bmw + gx] = 0;
  1022→        }
  1023→      }
  1024→    }
  1025→    // rubble is passable — route cost eases where the structure stood
  1026→    const c0x = Math.max(0, ((b.x - reach) / CELL) | 0);
  1027→    const c1x = Math.min(this.gw - 1, ((b.x + reach) / CELL) | 0);
  1028→    const c0y = Math.max(0, ((b.y - reach) / CELL) | 0);
  1029→    const c1y = Math.min(this.gh - 1, ((b.y + reach) / CELL) | 0);
  1030→    for (let gy = c0y; gy <= c1y; gy++) {
  1031→      for (let gx = c0x; gx <= c1x; gx++) {
  1032→        const i = gy * this.gw + gx;
  1033→        if (this.cost[i] > 0) this.cost[i] = Math.max(0.9, this.cost[i] - 5);
  1034→      }
  1035→    }
  1036→  }

  1038→  /** is the wall still standing where it meets (x, y)? */
  1039→  wallSegmentAlive(w: StoneWall, x: number, y: number): boolean {
  1040→    if (!w.segs || !w.segs.length) return true;
  1041→    const c = Math.cos(w.rot);
  1042→    const s = Math.sin(w.rot);
  1043→    const dx = x - w.x;
  1044→    const dy = y - w.y;
  1045→    const lx = dx * c + dy * s;
  1046→    const ly = -dx * s + dy * c;
  1047→    if (Math.abs(lx) > w.len / 2 + 4 || Math.abs(ly) > 12) return false;
  1048→    const segLen = w.len / w.segs.length;
  1049→    const idx = clamp(Math.floor((lx + w.len / 2) / segLen), 0, w.segs.length - 1);
  1050→    return w.segs[idx].hp > 0;
  1051→  }

  1053→  /** fraction of a wall still standing (0..1) */
  1054→  wallIntegrity(w: StoneWall): number {
  1055→    if (!w.segs || !w.segs.length) return 1;
  1056→    let up = 0;
  1057→    for (const s of w.segs) if (s.hp > 0) up++;
  1058→    return up / w.segs.length;
  1059→  }

  1063→  wallNear(x: number, y: number, r: number): StoneWall | null {
  1064→    let best: StoneWall | null = null;
  1065→    let bd = r;
  1066→    for (const w of this.walls) {
  1067→      const c = Math.cos(w.rot);
  1068→      const s = Math.sin(w.rot);
  1069→      const dx = x - w.x;
  1070→      const dy = y - w.y;
  1071→      const lx = dx * c + dy * s;
  1072→      const ly = -dx * s + dy * c;
  1073→      if (Math.abs(lx) > w.len / 2 + 4) continue;

  1075→      if (d < bd) {
  1076→        bd = d;
  1077→        best = w;

  1095→  private extractContours(): ContourSet {
  1096→    const step = 16; // metres between samples
  1097→    const nx = Math.floor(this.W / step) + 1;
  1098→    const ny = Math.floor(this.H / step) + 1;
  1099→    const heights = new Float32Array(nx * ny);
  1100→    for (let j = 0; j < ny; j++) {
  1101→      for (let i = 0; i < nx; i++) {
  1102→        heights[j * nx + i] = this.heightAt(i * step, j * step);
  1103→      }
  1104→    }
  1105→    const minor = new Path2D();
  1106→    const major = new Path2D();
  1107→    const interval = 8;
  1108→    const majorEvery = 5;
  1109→
  1110→    const lerpPt = (x1: number, y1: number, x2: number, y2: number, h1: number, h2: number, level: number) => {
  1111→      const t = (level - h1) / (h2 - h1 || 1e-6);
  1112→      return { x: (x1 + (x2 - x1) * t) * step, y: (y1 + (y2 - y1) * t) * step };
  1113→    };
  1114→
  1115→    for (let j = 0; j < ny - 1; j++) {
  1116→      for (let i = 0; i < nx - 1; i++) {
  1117→        const x = i * step;
  1118→        const y = j * step;
  1119→        const h00 = heights[j * nx + i];
  1120→        const h10 = heights[j * nx + i + 1];
  1121→        const h01 = heights[(j + 1) * nx + i];
  1122→        const h11 = heights[(j + 1) * nx + i + 1];
  1123→        const hMin = Math.min(h00, h10, h01, h11);
  1124→        const hMax = Math.max(h00, h10, h01, h11);
  1125→        const firstLevel = Math.ceil(hMin / interval) * interval;
  1126→        for (let level = firstLevel; level <= hMax; level += interval) {
  1127→          if (level <= 0) continue;
  1128→          const isMajor = Math.round(level / interval) % majorEvery === 0;
  1129→          const path = isMajor ? major : minor;
  1130→          // marching squares on cell corners
  1131→          let idx = 0;
  1132→          if (h00 >= level) idx |= 1;
  1133→          if (h10 >= level) idx |= 2;
  1134→          if (h11 >= level) idx |= 4;
  1135→          if (h01 >= level) idx |= 8;
  1136→          if (idx === 0 || idx === 15) continue;
  1137→          const top = () => lerpPt(i, j, i + 1, j, h00, h10, level);
  1138→          const right = () => lerpPt(i + 1, j, i + 1, j + 1, h10, h11, level);
  1139→          const bottom = () => lerpPt(i, j + 1, i + 1, j + 1, h01, h11, level);
  1140→          const left = () => lerpPt(i, j, i, j + 1, h00, h01, level);
  1141→          const seg = (a: { x: number; y: number }, b: { x: number; y: number }) => {
  1142→            path.moveTo(a.x, a.y);
  1143→            path.lineTo(b.x, b.y);
  1144→          };
  1145→          switch (idx) {
  1146→            case 1:
  1147→            case 14:
  1148→              seg(left(), top());
  1149→              break;
  1150→            case 2:
  1151→            case 13:
  1152→              seg(top(), right());
  1153→              break;
  1154→            case 3:
  1155→            case 12:
  1156→              seg(left(), right());
  1157→              break;
  1158→            case 4:
  1159→            case 11:
  1160→              seg(right(), bottom());
  1161→              break;
  1162→            case 6:
  1163→            case 9:
  1164→              seg(top(), bottom());
  1165→              break;
  1166→            case 7:
  1167→            case 8:
  1168→              seg(left(), bottom());
  1169→              break;
  1170→            case 5:
  1171→              seg(left(), top());
  1172→              seg(right(), bottom());
  1173→              break;
  1174→            case 10:
  1175→              seg(top(), right());
  1176→              seg(left(), bottom());
  1177→              break;
  1178→            default:
  1179→              break;
  1180→          }
  1181→        }
  1182→      }
  1183→    }
  1184→    return { minor, major };
  1185→  }

  1189→  private buildCostGrid() {
  1190→    this.gw = Math.ceil(this.W / CELL);
  1191→    this.gh = Math.ceil(this.H / CELL);
  1192→    const n = this.gw * this.gh;
  1193→    this.cost = new Float32Array(n);
  1194→    for (let gy = 0; gy < this.gh; gy++) {
  1195→      for (let gx = 0; gx < this.gw; gx++) {
  1196→        const cx = gx * CELL + CELL / 2;
  1197→        const cy = gy * CELL + CELL / 2;
  1198→        let c: number;
  1199→        if (this.isWater(cx, cy) && !this.bridgeAt(cx, cy, CELL * 0.9) && !this.fordAt(cx, cy, CELL)) {
  1200→          c = -1; // impassable water
  1201→        } else if (this.fordAt(cx, cy, CELL)) {
  1202→          c = 3.6; // wading — slow but possible
  1203→        } else {
  1204→          c = 1;
  1205→          c += this.slopeAt(cx, cy) * 14;
  1206→          const f = this.forestDensity(cx, cy);
  1207→          if (f > 0.42) c += 2.4;
  1208→          const r = this.roadFactor(cx, cy);
  1209→          if (r > 0.05) c = Math.min(c, r > 0.5 ? 0.34 : 0.6);
  1210→          if (this.railFactor(cx, cy) > 0.4) c += 1.2;
  1211→          if (this.buildingAt(cx, cy, 10)) c += 5;
  1212→          if (this.trenchDist(cx, cy) < 8) c += 1.4; // crossing a dug position
  1213→          // boulder fields and concrete are avoided by preference
  1214→          for (const rk of this.rocks) {
  1215→            if (rk.r < 2.6) continue;
  1216→            if (dist(cx, cy, rk.x, rk.y) < rk.r + 18) {
  1217→              c += 2.2;
  1218→              break;
  1219→            }
  1220→          }
  1221→          for (const b of this.barriers) {
  1222→            if ((b.hp ?? 100) <= 0) continue;
  1223→            if (dist(cx, cy, b.x, b.y) < 26) {
  1224→              c += 3.2;
  1225→              break;
  1226→            }
  1227→          }
  1228→          if (this.bridgeAt(cx, cy, CELL * 0.9)) c = 0.5;
  1229→        }
  1230→        this.cost[gy * this.gw + gx] = c;
  1231→      }
  1232→    }
  1233→  }

  1235→  private buildHeightGrid() {
  1236→    this.hgw = Math.ceil(this.W / HG_STEP) + 1;
  1237→    this.hgh = Math.ceil(this.H / HG_STEP) + 1;
  1238→    this.hg = new Float32Array(this.hgw * this.hgh);
  1239→    for (let j = 0; j < this.hgh; j++) {
  1240→      for (let i = 0; i < this.hgw; i++) {
  1241→        this.hg[j * this.hgw + i] = this.heightAt(i * HG_STEP, j * HG_STEP);
  1242→      }
  1243→    }
  1244→    // building occupancy mask — rasterize every footprint once
  1245→    this.bmw = Math.ceil(this.W / this.BM_CELL) + 1;
  1246→    this.bmh = Math.ceil(this.H / this.BM_CELL) + 1;
  1247→    this.bmask = new Uint8Array(this.bmw * this.bmh);
  1248→    // big boulders interrupt ground-level sightlines too
  1249→    this.rockMask = new Uint8Array(this.bmw * this.bmh);
  1250→    for (const rk of this.rocks) {
  1251→      if (rk.r < this.ROCK_LOS_R) continue;
  1252→      const g0x = Math.max(0, ((rk.x - rk.r) / this.BM_CELL) | 0);
  1253→      const g1x = Math.min(this.bmw - 1, ((rk.x + rk.r) / this.BM_CELL) | 0);
  1254→      const g0y = Math.max(0, ((rk.y - rk.r) / this.BM_CELL) | 0);
  1255→      const g1y = Math.min(this.bmh - 1, ((rk.y + rk.r) / this.BM_CELL) | 0);
  1256→      for (let gy = g0y; gy <= g1y; gy++) {
  1257→        for (let gx = g0x; gx <= g1x; gx++) {
  1258→          const px = gx * this.BM_CELL + this.BM_CELL / 2;
  1259→          const py = gy * this.BM_CELL + this.BM_CELL / 2;
  1260→          if (dist(px, py, rk.x, rk.y) < rk.r * 0.9) this.rockMask[gy * this.bmw + gx] = 1;
  1261→        }
  1262→      }
  1263→    }
  1264→    for (const b of this.buildings) {
  1265→      if (b.kind === 'MAST' || b.kind === 'CHECKPOINT') continue; // see-through
  1266→      const c = Math.cos(b.rot);
  1267→      const s = Math.sin(b.rot);
  1268→      // intersect-based rasterisation: any cell the footprint touches is
  1269→      // masked, so even a small house reliably interrupts a ground beam
  1270→      const reach = Math.max(b.w, b.h) * 0.5;
  1271→      const gx0 = Math.max(0, ((b.x - reach) / this.BM_CELL) | 0);
  1272→      const gx1 = Math.min(this.bmw - 1, ((b.x + reach) / this.BM_CELL) | 0);
  1273→      const gy0 = Math.max(0, ((b.y - reach) / this.BM_CELL) | 0);
  1274→      const gy1 = Math.min(this.bmh - 1, ((b.y + reach) / this.BM_CELL) | 0);
  1275→      for (let gy = gy0; gy <= gy1; gy++) {
  1276→        for (let gx = gx0; gx <= gx1; gx++) {
  1277→          // closest point of the cell rect to the building centre, in
  1278→          // building-local space — if inside the half-extents, mask
  1279→          const px = clamp(b.x, gx * this.BM_CELL, (gx + 1) * this.BM_CELL);
  1280→          const py = clamp(b.y, gy * this.BM_CELL, (gy + 1) * this.BM_CELL);
  1281→          const dx = px - b.x;
  1282→          const dy = py - b.y;
  1283→          const lx = dx * c + dy * s;
  1284→          const ly = -dx * s + dy * c;
  1285→          if (Math.abs(lx) <= b.w / 2 && Math.abs(ly) <= b.h / 2) {
  1286→            this.bmask[gy * this.bmw + gx] = 1;
  1287→          }
  1288→        }
  1289→      }
  1290→    }
  1291→  }

  1293→  findPath(from: Vec2, to: Vec2): Vec2[] {
  1294→    const sgx = clamp(Math.floor(from.x / CELL), 0, this.gw - 1);
  1295→    const sgy = clamp(Math.floor(from.y / CELL), 0, this.gh - 1);
  1296→    let tgx = clamp(Math.floor(to.x / CELL), 0, this.gw - 1);
  1297→    let tgy = clamp(Math.floor(to.y / CELL), 0, this.gh - 1);
  1298→    if (this.cost[tgy * this.gw + tgx] < 0) {
  1299→      // nudge target to nearest passable cell
  1300→      const alt = this.nearestPassable(tgx, tgy);
  1301→      if (!alt) return [{ ...to }];
  1302→      tgx = alt[0];
  1303→      tgy = alt[1];
  1304→    }
  1305→    if (sgx === tgx && sgy === tgy) return [{ ...to }];
  1306→
  1307→    const n = this.gw * this.gh;
  1308→    const gScore = new Float32Array(n).fill(Infinity);
  1309→    const cameFrom = new Int32Array(n).fill(-1);
  1310→    const closed = new Uint8Array(n);
  1311→    const open: number[] = [];
  1312→    const fScore = new Float32Array(n).fill(Infinity);
  1313→    const h = (x: number, y: number) => (Math.abs(x - tgx) + Math.abs(y - tgy)) * 0.6;
  1314→    const start = sgy * this.gw + sgx;
  1315→    gScore[start] = 0;
  1316→    fScore[start] = h(sgx, sgy);
  1317→    open.push(start);
  1318→    let guard = 0;
  1319→    while (open.length && guard++ < 12000) {
  1320→      // find best in open
  1321→      let bi = 0;
  1322→      for (let i = 1; i < open.length; i++) {
  1323→        if (fScore[open[i]] < fScore[open[bi]]) bi = i;
  1324→      }
  1325→      const cur = open.splice(bi, 1)[0];
  1326→      if (cur === tgy * this.gw + tgx) break;
  1327→      closed[cur] = 1;
  1328→      const cx = cur % this.gw;
  1329→      const cy = (cur / this.gw) | 0;
  1330→      for (let dy = -1; dy <= 1; dy++) {
  1331→        for (let dx = -1; dx <= 1; dx++) {
  1332→          if (!dx && !dy) continue;
  1333→          const nx = cx + dx;
  1334→          const ny = cy + dy;
  1335→          if (nx < 0 || ny < 0 || nx >= this.gw || ny >= this.gh) continue;
  1336→          const ni = ny * this.gw + nx;
  1337→          if (closed[ni] || this.cost[ni] < 0) continue;
  1338→          const step = (dx && dy ? 1.414 : 1) * (this.cost[ni] + this.cost[cur]) * 0.5;
  1339→          const tentative = gScore[cur] + step;
  1340→          if (tentative < gScore[ni]) {
  1341→            gScore[ni] = tentative;
  1342→            cameFrom[ni] = cur;
  1343→            fScore[ni] = tentative + h(nx, ny);
  1344→            if (!open.includes(ni)) open.push(ni);
  1345→          }
  1346→        }
  1347→      }
  1348→    }
  1349→
  1350→    const end = tgy * this.gw + tgx;
  1351→    if (cameFrom[end] === -1 && end !== start) return [{ ...to }];
  1352→    // reconstruct
  1353→    const cells: number[] = [];
  1354→    let c = end;
  1355→    while (c !== -1 && c !== start) {
  1356→      cells.push(c);
  1357→      c = cameFrom[c];
  1358→    }
  1359→    cells.push(start);
  1360→    cells.reverse();
  1361→    // waypoints at cell centres, then string-pull
  1362→    let pts: Vec2[] = cells.map((ci) => ({
  1363→      x: (ci % this.gw) * CELL + CELL / 2,
  1364→      y: ((ci / this.gw) | 0) * CELL + CELL / 2,
  1365→    }));
  1366→    pts.push({ ...to });
  1367→    pts = this.smoothPath(pts);
  1368→    return pts.length ? pts : [{ ...to }];
  1369→  }

  1371→  private nearestPassable(gx: number, gy: number): [number, number] | null {
  1372→    for (let r = 1; r < 8; r++) {
  1373→      for (let dy = -r; dy <= r; dy++) {
  1374→        for (let dx = -r; dx <= r; dx++) {
  1375→          const nx = gx + dx;
  1376→          const ny = gy + dy;
  1377→          if (nx < 0 || ny < 0 || nx >= this.gw || ny >= this.gh) continue;
  1378→          if (this.cost[ny * this.gw + nx] >= 0) return [nx, ny];
  1379→        }
  1380→      }
  1381→    }
  1382→    return null;
  1383→  }

  1385→  private smoothPath(pts: Vec2[]): Vec2[] {
  1386→    if (pts.length < 3) return pts;
  1387→    const out: Vec2[] = [pts[0]];
  1388→    let i = 0;
  1389→    while (i < pts.length - 1) {
  1390→      let j = pts.length - 1;
  1391→      for (; j > i + 1; j--) {
  1392→        if (this.los(pts[i], pts[j])) break;
  1393→      }
  1394→      out.push(pts[j]);
  1395→      i = j;
  1396→    }
  1397→    return out;
  1398→  }

  1400→  los(a: Vec2, b: Vec2): boolean {
  1401→    const d = dist(a.x, a.y, b.x, b.y);
  1402→    const steps = Math.max(2, Math.ceil(d / (CELL * 0.6)));
  1403→    for (let s = 1; s < steps; s++) {
  1404→      const t = s / steps;
  1405→      const x = a.x + (b.x - a.x) * t;
  1406→      const y = a.y + (b.y - a.y) * t;
  1407→      if (this.isWater(x, y) && !this.bridgeAt(x, y, 30) && !this.fordAt(x, y, 40)) return false;
  1408→      if (this.slopeAt(x, y) > 0.75) return false;
  1409→    }
  1410→    return true;
  1411→  }

  1434→  distToPolyline(x: number, y: number, pts: Vec2[]): number {
  1435→    let best = Infinity;
  1436→    for (let i = 0; i < pts.length - 1; i++) {
  1437→      const a = pts[i];
  1438→      const b = pts[i + 1];
  1439→      const vx = b.x - a.x;
  1440→      const vy = b.y - a.y;
  1441→      const wx = x - a.x;
  1442→      const wy = y - a.y;
  1443→      const len2 = vx * vx + vy * vy || 1e-6;
  1444→      let t = (wx * vx + wy * vy) / len2;
  1445→      t = clamp(t, 0, 1);
  1446→      const dx = x - (a.x + vx * t);
  1447→      const dy = y - (a.y + vy * t);
  1448→      const d = dx * dx + dy * dy;
  1449→      if (d < best) best = d;
  1450→    }
  1451→    return Math.sqrt(best);
  1452→  }

  1454→  polylineDirAt(x: number, y: number, pts: Vec2[]): number {
  1455→    let best = Infinity;
  1456→    let bestI = 0;
  1457→    let bestT = 0;
  1458→    for (let i = 0; i < pts.length - 1; i++) {
  1459→      const a = pts[i];
  1460→      const b = pts[i + 1];
  1461→      const vx = b.x - a.x;
  1462→      const vy = b.y - a.y;
  1463→      const wx = x - a.x;
  1464→      const wy = y - a.y;
  1465→      const len2 = vx * vx + vy * vy || 1e-6;
  1466→      let t = (wx * vx + wy * vy) / len2;
  1467→      t = clamp(t, 0, 1);
  1468→      const dx = x - (a.x + vx * t);
  1469→      const dy = y - (a.y + vy * t);
  1470→      const d = dx * dx + dy * dy;
  1471→      if (d < best) {
  1472→        best = d;
  1473→        bestI = i;
  1474→        bestT = t;
  1475→      }
  1476→    }
  1477→    const a = pts[bestI];
  1478→    const b = pts[bestI + 1];
  1479→    void bestT;
  1480→    return Math.atan2(b.y - a.y, b.x - a.x);
  1481→  }
